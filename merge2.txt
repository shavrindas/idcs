from functools import lru_cache
from pydantic_settings import BaseSettings, SettingsConfigDict

class Settings(BaseSettings):
    ENABLE_FACILITY_SERVER: bool = False
    ENABLE_CENTER_SERVER: bool = False
    ENABLE_WEB_BACKEND: bool = False
    ENABLE_APP_BACKEND: bool = False
    ENABLE_DEBUG_SETUP: bool = False

    # 실행 루트(idc_server_v2) 기준의 .env를 읽도록 설정
    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        extra="ignore"
    )

@lru_cache
def get_settings() -> Settings:
    return Settings()
