    """Box #1: 일반 계정 생성 (비동기 처리)"""

    result = await db.execute(select(User).where(User.email == link_in.email))
    user = result.scalars().first()
    if not user:
        raise HTTPException(status_code=404, detail="사용자를 찾을 수 없습니다.")

    existing_provider_result = await db.execute(
        select(ExternalProvider).where(
            ExternalProvider.account_id == user.id,
            ExternalProvider.provider == link_in.provider
        )
    )
    if existing_provider_result.scalars().first():
        raise HTTPException(status_code=400, detail="이미 해당 소셜 계정이 연동되어 있습니다.")

    new_provider = ExternalProvider(
        account_id=user.id,
        provider=link_in.provider,
        provider_id=link_in.provider_id
    )
    db.add(new_provider)
    await db.commit()

    return {"message": f"계정에 {link_in.provider} 연동이 완료되었습니다."}

    """인증 코드를 해싱합니다."""

@router.post("/password/verify-code", status_code=status.HTTP_200_OK)
async def verify_password_code(req: PasswordVerifyCode):
    """Box #2: 입력 코드 확인 및 재설정 토큰 발급"""
    try:
        payload = jwt.decode(req.token, SECRET_KEY, algorithms=[ALGORITHM])
        email = payload.get("sub")
        code_hash = payload.get("code_hash")

        # 클라이언트가 보낸 코드를 해싱하여 토큰의 해시값과 비교
        if hash_code(req.code) != code_hash:
            raise HTTPException(status_code=400, detail="코드가 일치하지 않습니다.")

        # 인증 성공 시, 비밀번호 변경 전용 새 JWT 발급 (만료시간 10분)
        expire = datetime.now(timezone.utc) + timedelta(minutes=10)
        reset_token = jwt.encode(
            {"sub": email, "action": "reset_pw", "exp": expire}, 
            SECRET_KEY, 
            algorithm=ALGORITHM
        )

        return {"message": "코드 인증이 완료되었습니다.", "reset_token": reset_token}
        
    except jwt.ExpiredSignatureError:
        raise HTTPException(status_code=400, detail="인증 시간이 초과되었습니다. 코드를 다시 요청해주세요.")
    except jwt.InvalidTokenError:
        raise HTTPException(status_code=400, detail="유효하지 않은 토큰입니다.")

        FROM posts p
        JOIN boards b ON p.board_id = b.board_id
        JOIN user_table u ON p.creator_id = u.id
        {where_clause}
    """)

@router.put("/post/{post_id}")
async def update_post(post_id: int, user_id: int, req: PostUpdateReq, db: AsyncSession = Depends(get_db)):
    # 1. 작성자 본인 확인
    check_query = text("SELECT creator_id FROM posts WHERE post_id = :post_id")
    post = (await db.execute(check_query, {"post_id": post_id})).fetchone()
    if not post or post.creator_id != user_id:
        raise HTTPException(status_code=403, detail="수정 권한이 없습니다.")

    # 2. 포스트 업데이트
    update_query = text("""
        UPDATE posts 
        SET title = COALESCE(:title, title),
            content = COALESCE(:content, content),
            updater_id = :user_id
        WHERE post_id = :post_id
    """)
    await db.execute(update_query, {
        "title": req.title, 
        "content": req.content, 
        "user_id": user_id, 
        "post_id": post_id
    })

    # 3. 추가된 이미지가 있다면 맵핑
    if req.image_ids:
        mapping_query = text("""
            UPDATE post_images 
            SET post_id = :post_id 
            WHERE image_id = ANY(:image_ids)
        """)
        await db.execute(mapping_query, {"post_id": post_id, "image_ids": req.image_ids})

    await db.commit()
    return {"status": "success", "message": "포스트 수정 완료"}

from fastapi import APIRouter, Depends, Query, HTTPException
from typing import Optional, Dict, Any, List
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import text
from datetime import datetime, timedelta
from config.db import get_db

router = APIRouter(prefix="/chart", tags=["Chart"])

SCALE_MAPPING = {
    "10s": "10 seconds",
    "1m": "1 minute",
    "5m": "5 minutes",
    "10m": "10 minutes",  
    "15m": "15 minutes",  
    "1h": "1 hour"
}


@router.get("/{facility_id}/{room_id}")
async def get_chart_data(
    facility_id: int,
    room_id: int,
    scale: str = Query("1m", description="데이터 해상도"),
    start_time: Optional[datetime] = Query(None, description="조회 시작 시간"),
    end_time: Optional[datetime] = Query(None, description="조회 종료 시간"),
    role_ids: str = Query(..., description="조회할 PointRole ID 목록"),
    group_id: Optional[int] = Query(None, description="현재 활성화된 차트 그룹 ID (블랙리스트 조회용)"),
    db: AsyncSession = Depends(get_db)
):
    
    if scale not in SCALE_MAPPING:
        raise HTTPException(status_code=400, detail="유효하지 않은 스케일입니다.")
    
    bucket_interval = SCALE_MAPPING[scale]
    target_roles = [int(r.strip()) for r in role_ids.split(",")]

    role_query = text("""
        SELECT pr.id, pr.role_name, pr.target_value, pr.target_low_limit, pr.target_high_limit, prm.data_type
        FROM point_role_table pr
        LEFT JOIN point_role_metadata prm ON pr.id = prm.role_id
        WHERE pr.id = ANY(:role_ids) AND pr.is_current = TRUE
    """)
    role_result = await db.execute(role_query, {"role_ids": target_roles})
    roles_info = {row.id: dict(row._mapping) for row in role_result.fetchall()}

    if not roles_info:
        return {"status": "success", "data": {"datasets": [], "alarms": [], "posts": [], "hidden_posts": []}, "message": "유효한 Role 데이터가 없습니다."}

    temp_roles, gear_roles, static_roles, bool_roles = [], [], [], []
    for r_id, r_info in roles_info.items():
        dtype = r_info.get("data_type")
        
        # 메타데이터가 아직 할당되지 않은 예전 롤에 대한 방어 로직 (이름 기반 Fallback)
        if not dtype:
            r_name = r_info["role_name"].lower()
            if "temp_" in r_name: temp_roles.append(r_id)
            elif "gear_" in r_name: gear_roles.append(r_id)
            elif "static_" in r_name: static_roles.append(r_id)
            elif "bool_" in r_name: bool_roles.append(r_id)
            continue
            
        if dtype == 'BOOL':
            bool_roles.append(r_id)
        elif dtype == 'STATIC':
            static_roles.append(r_id)
        elif dtype == 'GEAR':
            gear_roles.append(r_id) # 쿼리는 gear 테이블에서 수행 (결과는 메인차트로 감)
        else: # TEMP 및 기본
            temp_roles.append(r_id)

    if not end_time:
        max_time_query = text("SELECT MAX(timestamp) FROM temperature_numerical_logs WHERE role_id = ANY(:role_ids)")
        end_time = (await db.execute(max_time_query, {"role_ids": temp_roles if temp_roles else target_roles})).scalar()
        if not end_time:
            return {"status": "success", "data": {"datasets": [], "alarms": [], "posts": [], "hidden_posts": []}, "message": "기록된 로그가 없습니다."}

    if not start_time:
        start_time = end_time - timedelta(days=1)

    datasets_map: Dict[int, Any] = {
        r_id: {
            "role_id": r_id, "role_name": roles_info[r_id]["role_name"],
            "target_value": roles_info[r_id]["target_value"],
            "target_low_limit": roles_info[r_id]["target_low_limit"],
            "target_high_limit": roles_info[r_id]["target_high_limit"],
            "data": []
        } for r_id in roles_info.keys()
    }

    async def fetch_and_map(table_name: str, agg_columns: str, type_roles: List[int], data_key_mapping: dict):
        if not type_roles: return
        query = text(f"""
            SELECT role_id, time_bucket('{bucket_interval}', timestamp) AS bucket, {agg_columns}
            FROM {table_name}
            WHERE role_id = ANY(:role_ids)
              AND timestamp >= :start_time
              AND timestamp <= :end_time
            GROUP BY role_id, bucket ORDER BY bucket ASC
        """)
        rows = await db.execute(query, {"role_ids": type_roles, "start_time": start_time, "end_time": end_time})
        for r in rows.fetchall():
            r_dict = dict(r._mapping)
            point = {"timestamp": r_dict["bucket"].isoformat()}
            for db_key, out_key in data_key_mapping.items(): point[out_key] = r_dict[db_key]
            datasets_map[r_dict["role_id"]]["data"].append(point)

    await fetch_and_map("temperature_numerical_logs", "AVG(temp_mean) AS mean, MIN(temp_min) AS min, MAX(temp_max) AS max", temp_roles, {"mean": "mean", "min": "min", "max": "max"})
    await fetch_and_map("gear_numerical_logs", "AVG(gear_mean) AS mean, MIN(gear_min) AS min, MAX(gear_max) AS max", gear_roles, {"mean": "mean", "min": "min", "max": "max"})
    await fetch_and_map("equipment_static_logs", "AVG(setnum) AS mean", static_roles, {"mean": "mean"})
    await fetch_and_map("bool_logs", "BOOL_OR(status_value) AS status", bool_roles, {"status": "status"})

    # 💡 [신규 추가] 알람(Alarm) 데이터 조회
    alarm_query = text("""
        SELECT alarm_id, role_id, alarm_name, severity, timestamp, is_triggered, end_time
        FROM alarms
        WHERE role_id = ANY(:role_ids)
          AND timestamp >= :start_time
          AND timestamp <= :end_time
        ORDER BY timestamp ASC
    """)
    alarm_rows = await db.execute(alarm_query, {"role_ids": target_roles, "start_time": start_time, "end_time": end_time})
    alarms_list = []
    for row in alarm_rows.fetchall():
        alarm_dict = dict(row._mapping)
        if alarm_dict.get("timestamp"):
            alarm_dict["timestamp"] = alarm_dict["timestamp"].isoformat()
        if alarm_dict.get("end_time"):
            alarm_dict["end_time"] = alarm_dict["end_time"].isoformat()
        alarms_list.append(alarm_dict)

    # 1. 시설에 해당하는 포스트 중 기간이 설정되어 있고 차트 조회 범위와 겹치는 항목 조회
    post_query = text("""
        SELECT p.post_id as id, p.title, p.content, p.start_date, p.end_date
        FROM posts p
        JOIN boards b ON p.board_id = b.board_id
        WHERE b.facility_id = :facility_id
          AND p.start_date IS NOT NULL 
          AND p.end_date IS NOT NULL
          /* 💡 [수정됨] DATE 타입을 TIMESTAMP WITH TIME ZONE으로 캐스팅하여 타입 충돌 방어 */
          AND CAST(p.start_date AS TIMESTAMP WITH TIME ZONE) <= :end_time
          AND (CAST(p.end_date AS TIMESTAMP WITH TIME ZONE) + INTERVAL '1 day') > :start_time
    """)
    post_rows = await db.execute(post_query, {"facility_id": facility_id, "start_time": start_time, "end_time": end_time})
    all_posts = [dict(r._mapping) for r in post_rows.fetchall()]
    
    # 2. 현재 차트 그룹의 숨겨진 포스트 ID 목록 조회
    hidden_post_ids = set()
    if group_id:
        hidden_query = text("SELECT post_id FROM post_hidden_chart_groups WHERE group_id = :group_id")
        hidden_rows = await db.execute(hidden_query, {"group_id": group_id})
        hidden_post_ids = {r[0] for r in hidden_rows.fetchall()}

    visible_posts = []
    hidden_posts = []
    for p in all_posts:
        p_dict = dict(p)
        if p_dict.get("start_date"): p_dict["start_date"] = p_dict["start_date"].isoformat()
        if p_dict.get("end_date"): p_dict["end_date"] = p_dict["end_date"].isoformat()
        
        if p_dict["id"] in hidden_post_ids:
            hidden_posts.append(p_dict)
        else:
            visible_posts.append(p_dict)

    # 💡 최종 통합 응답 반환
    return {
        "status": "success",
        "meta": {"scale": scale, "start_time": start_time.isoformat(), "end_time": end_time.isoformat()},
        "data": {
            "datasets": list(datasets_map.values()),
            "alarms": alarms_list,
            "posts": visible_posts,
            "hidden_posts": hidden_posts
        }
    }

class ConnectionManager:
    def __init__(self):
        self.active_connections: Dict[int, List[WebSocket]] = {}

    async def connect(self, websocket: WebSocket, room_id: int):
        await websocket.accept()
        if room_id not in self.active_connections:
            self.active_connections[room_id] = []
        self.active_connections[room_id].append(websocket)

    def disconnect(self, websocket: WebSocket, room_id: int):
        if room_id in self.active_connections:
            self.active_connections[room_id].remove(websocket)
            if not self.active_connections[room_id]:
                del self.active_connections[room_id]

    async def broadcast_to_room(self, room_id: int, message: dict):
        if room_id in self.active_connections:
            for connection in self.active_connections[room_id]:
                await connection.send_text(json.dumps(message))

manager = ConnectionManager()

    processed_rooms.sort(key=lambda x: x["last_message_time"], reverse=True)

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import text
from config.db import get_db

router = APIRouter(prefix="/api/test-auth", tags=["Test Auth"])



@router.get("/my-context/{user_id}")
async def get_my_context(user_id: int, db: AsyncSession = Depends(get_db)):
    # 1. 유저 정보 조회
    user_query = text("SELECT id, email, name, role FROM user_table WHERE id = :user_id")
    user_result = await db.execute(user_query, {"user_id": user_id})
    user = user_result.fetchone()

    if not user:
        raise HTTPException(status_code=404, detail="User not found")

    # 2. 시설 정보 조회 로직
    # 관리자(ADMIN, DEVELOPER)는 모든 시설에 접근 가능
    if user.role in ['ADMIN', 'DEVELOPER']:
        facility_query = text("SELECT id, name FROM facilities_table WHERE is_active = TRUE")
        facility_result = await db.execute(facility_query)
    else:
        # 일반 유저(USER)는 facility_users 매핑 테이블에 존재하는 시설만 접근 가능
        facility_query = text("""
            SELECT f.id, f.name 
            FROM facilities_table f
            JOIN facility_users fu ON f.id = fu.facility_id
            WHERE fu.user_id = :user_id AND f.is_active = TRUE
        """)
        facility_result = await db.execute(facility_query, {"user_id": user_id})
        
    facilities = [{"facility_id": row.id, "facility_name": row.name} for row in facility_result.fetchall()]

    return {
        "status": "success",
        "user": {
            "id": user.id,
            "email": user.email,
            "name": user.name,
            "role": user.role
        },
        "accessible_facilities": facilities,
        # 임시로 클라이언트가 기본으로 선택할 첫 번째 시설 ID 반환 (없으면 None)
        "default_facility_id": facilities[0]["facility_id"] if facilities else None
    }

import asyncio
from firebase_admin import messaging

async def send_multicast_fcm(tokens: list[str], title: str, body: str):
    if not tokens:
        return

    message = messaging.MulticastMessage(
        notification=messaging.Notification(title=title, body=body),
        tokens=tokens, # 토큰 리스트 [token1, token2, ...]
    )
    
    # 한번에 최대 500개 토큰까지 전송 가능
    response = await asyncio.to_thread(messaging.send_each_for_multicast, message)
    print(f"[FCM 다중 전송] 성공: {response.success_count}건, 실패: {response.failure_count}건")

#캐시 시간
CACHE_TTL = 300

role_cache: dict[int, list] = {}

def invalidate_role_cache(facility_id: int):
    role_cache.pop(f"{facility_id}_list", None)
    role_cache.pop(f"{facility_id}_dict", None)


# 2단계: 기존 리스트 형태의 함수 (캐시 키에 _list 추가)
async def scan_roles(facility_id: int, db: AsyncSession = Depends(get_db)):
    now = time.time()
    cache_entry = role_cache.get(facility_id)
    if not cache_entry or (now - cache_entry["updated_at"] > CACHE_TTL):
        async for db in get_db():
            role_result = await db.execute(
                select(PointRole).where(PointRole.facility_id == facility_id)
            )
            role_cache[facility_id] = {
                "roles": role_result.scalars().all(),
                "updated_at": now
            }
            #print(f"[DB 갱신] Facility {facility_id} PointRole 캐시 업데이트 완료")
            break
        
    roles = role_cache[facility_id]["roles"]
    return roles


# 3단계: 딕셔너리 형태의 함수 (캐시 키에 _dict 추가)
async def scan_roles_asc_name(facility_id: int, db: AsyncSession = Depends(get_db)):
    now = time.time()
    cache_key = f"{facility_id}_dict" # 🟢 키 분리
    cache_entry = role_cache.get(cache_key)
    
    if (not cache_entry or 
        (now - cache_entry["updated_at"] > CACHE_TTL)):
        
        async for db in get_db():
            role_result = await db.execute(
                select(PointRole).where(PointRole.facility_id == facility_id)
            )
            
        raw_roles = role_result.scalars().all()
        
        sorted_roles = sorted(
            raw_roles, 
            key=lambda x: f"{x.room_id}.{x.role_name}"
        )
        
        roles_dict = {
            f"{role.room_id}.{role.role_name}": role 
            for role in sorted_roles
        }
        
        role_cache[cache_key] = {
            "roles": roles_dict,
            "updated_at": now
        }
        
    roles = role_cache[cache_key]["roles"]
    return roles

async def check_roles(facility_id: int, data_dict: Dict[str, str], roles):
    tokens: List[str] = []
    # 단순 센서키 대신 (센서키, matching_role) 튜플을 저장
    triggered_alarms: List[tuple] = []
    facility_name: str = f"시설 #{facility_id}" # 기본값 설정

    async for db in get_db():
        candidate_alarms = []
        # 1. 알람 대상 조건 검사 및 발생 대상 선별
        for role in roles:
            for k, v in data_dict.items():
                if role.role_name in k and str(role.room_id) in k:
                    val = float(v)
                    low_limit = float(role.target_low_limit) if role.target_low_limit not in (None, "", "null") else float('-inf')
                    high_limit = float(role.target_high_limit) if role.target_high_limit not in (None, "", "null") else float('inf')

                    is_out_of_bounds = val <= low_limit or val >= high_limit
                    
                    if is_out_of_bounds:
                        candidate_alarms.append((k, role))

        # 2. 이상이 발생한 항목이 있을 때만 단 1번의 DB 쿼리로 기존 알람들 조회 (In-Memory 매핑)
        if candidate_alarms:
            # 검사할 알람명 목록 추출
            candidate_names = [AMM.Maped.get(k, k) for k, _ in candidate_alarms]

            # IN 쿼리로 한 번에 조회
            stmt = select(Alarm).where(Alarm.alarm_name.in_(candidate_names))
            result = await db.execute(stmt)
    
            # {alarm_name: Alarm 객체} 형태의 딕셔너리로 변환
            existing_alarms = {a.alarm_name: a for a in result.scalars().all()}

            now_utc = datetime.now(timezone.utc)

            # 3. end_time이 지나지 않은(진행 중인) 알람 필터링
            for k, role in candidate_alarms:
                am_name = AMM.Maped.get(k, k)
                mapping = existing_alarms.get(am_name)

                # 이미 활성화되어 진행 중인 알람이면 제외
                if mapping and mapping.end_time and (mapping.end_time >= now_utc):
                    continue

                # 신규 발생 대상에 추가
                triggered_alarms.append((k, role))

        # 2. 발생 대상이 있을 경우: DB에 Alarm 기록 저장 & FCM 토큰 조회
        if triggered_alarms:
            facility_stmt = select(Facility.name).where(Facility.id == facility_id)
            facility_res = await db.execute(facility_stmt)
            fetched_name = facility_res.scalar_one_or_none()
            if fetched_name:
                facility_name = fetched_name

            now_time = datetime.now(timezone.utc)
            
            # 2-1. DB에 신규 Alarm 인스턴스 기록
            for k, role in triggered_alarms:
                am_name = AMM.Maped.get(k, k)
                
                new_alarm = Alarm(
                    role_id=role.id,                         # matching PointRole의 PK ID
                    alarm_name=am_name,                      # 알람명
                    severity=getattr(role, 'severity', 'a'), # role에 설정된 severity 사용 (기본값 'a')
                    timestamp=now_time,                      # 발생 시간
                    is_triggered=True,                       # 알람 발동 상태
                    reset_timeout_interval=timedelta(hours=24), # 기본 리셋 타임아웃
                    end_time=now_time + timedelta(hours=24)  # end_time 기본 24시간 지정 예시
                )
                db.add(new_alarm)
            
            # DB 변경사항 적용
            await db.commit()

            # 2-2. FCM 토큰 1회 조회 (JOIN)
            stmt_token = (
                select(UserFcmToken.fcm_token)
                .join(FacilityUser, FacilityUser.user_id == UserFcmToken.user_id)
                .where(FacilityUser.facility_id == facility_id)
                .distinct()
            )
            res_token = await db.execute(stmt_token)
            tokens = list(res_token.scalars().all())

        break # get_db() 생성기 1회 수행 후 탈출

    # 3. FCM 푸시 알림 전송
    if tokens and triggered_alarms:
        print("알람 발생 : ", [k for k, _ in triggered_alarms])
        for k, _ in triggered_alarms:
            mapped_name = AMM.Maped.get(k)
            if mapped_name:
                await msg.send_multicast_fcm(tokens, facility_name + "이상 발생", mapped_name)
            else:
                await msg.send_multicast_fcm(tokens, facility_name + "이상 발생", f"{k}")

const SYNC_GROUP_NAME = 'idcool-chart-group';
echarts.connect(SYNC_GROUP_NAME);

const formatShortTime = (ts: number) => {
  const d = new Date(ts);
  return `${(d.getMonth() + 1).toString().padStart(2, '0')}/${d.getDate().toString().padStart(2, '0')} ${d.getHours().toString().padStart(2, '0')}:${d.getMinutes().toString().padStart(2, '0')}`;
};

const generatePostSegments = (validPosts: any[]) => {
  if (!validPosts || validPosts.length === 0) return [];
  const points = new Set<number>();
  validPosts.forEach(p => { points.add(p.st); points.add(p.et); });
  const sortedPoints = Array.from(points).sort((a, b) => a - b);
  const segments: any[] = [];
  
  for (let i = 0; i < sortedPoints.length - 1; i++) {
    const segSt = sortedPoints[i];
    const segEt = sortedPoints[i + 1];
    if (segSt === segEt) continue;
    const activePosts = validPosts.filter(p => p.st <= segSt && p.et >= segEt);
    if (activePosts.length > 0) segments.push({ st: segSt, et: segEt, activePosts });
  }
  return segments;
};

import { useState, useEffect } from 'react';
import { apiService } from '../services/apiService';
import { LeftPanel } from '../components/chart/mapper/LeftPanel';
import { RightPanel } from '../components/chart/mapper/RightPanel';
import type { Role, ChartGroup } from '../types';

export const ChartMappingPage: React.FC = () => {
  const [facilities, setFacilities] = useState<any[]>([]);
  const [rooms, setRooms] = useState<any[]>([]);
  const [selectedFacility, setSelectedFacility] = useState<number | null>(null);
  const [selectedRoom, setSelectedRoom] = useState<number | null>(null);

  const [room0Roles, setRoom0Roles] = useState<Role[]>([]);
  const [targetRoomRoles, setTargetRoomRoles] = useState<Role[]>([]);

  const [existingGroups, setExistingGroups] = useState<ChartGroup[]>([]);
  const [selectedGroupId, setSelectedGroupId] = useState<number | 'new'>('new');
  const [groupName, setGroupName] = useState('');
  const [droppedRoles, setDroppedRoles] = useState<Role[]>([]);

  useEffect(() => {
    apiService.getFacilities().then((data) => setFacilities(data.facilities || []));
  }, []);

  const handleFacilityChange = async (facId: number) => {
    setSelectedFacility(facId);
    setSelectedRoom(null);
    setTargetRoomRoles([]);
    setExistingGroups([]);
    resetGroupForm();

    const roomData = await apiService.getRooms(facId);
    setRooms(roomData.rooms || []);
    const roleData = await apiService.getPointRoles(facId, 0);
    setRoom0Roles(roleData.roles || []);
  };

  const handleRoomChange = async (roomId: number) => {
    setSelectedRoom(roomId);
    resetGroupForm();
    if (selectedFacility) {
      const roleData = await apiService.getPointRoles(selectedFacility, roomId);
      setTargetRoomRoles(roleData.roles || []);
      const groupData = await apiService.getChartGroups(selectedFacility, roomId);
      setExistingGroups(groupData.groups || []);
    }
  };

  const resetGroupForm = () => {
    setSelectedGroupId('new');
    setGroupName('');
    setDroppedRoles([]);
  };

  const handleGroupSelect = (val: string) => {
    if (val === 'new') {
      resetGroupForm();
      return;
    }
    const gId = Number(val);
    setSelectedGroupId(gId);
    const targetGroup = existingGroups.find((g) => g.group_id === gId);
    if (targetGroup) {
      setGroupName(targetGroup.group_name);
      const allAvailableRoles = [...room0Roles, ...targetRoomRoles];
      const validRoleIds = (targetGroup.role_ids || []).filter((id) => id !== null);
      const loadedRoles = validRoleIds
        .map((id) => allAvailableRoles.find((r) => r.role_id === id))
        .filter(Boolean) as Role[];
      setDroppedRoles(loadedRoles);
    }
  };

  const onDragStart = (e: React.DragEvent, role: Role) => {
    e.dataTransfer.setData('application/json', JSON.stringify(role));
  };

  const onDragOver = (e: React.DragEvent) => {
    e.preventDefault();
  };

  const onDrop = (e: React.DragEvent) => {
    e.preventDefault();
    const roleData = e.dataTransfer.getData('application/json');
    if (roleData) {
      const role: Role = JSON.parse(roleData);
      if (!droppedRoles.find((r) => r.role_id === role.role_id)) {
        setDroppedRoles([...droppedRoles, role]);
      }
    }
  };

  const handleReorderRoles = (sourceIndex: number, targetIndex: number) => {
    if (sourceIndex === targetIndex) return;
    const updated = [...droppedRoles];
    const [movedItem] = updated.splice(sourceIndex, 1);
    updated.splice(targetIndex, 0, movedItem);
    setDroppedRoles(updated);
  };

  const handleSaveGroup = async () => {
    if (!selectedFacility || !selectedRoom) return alert('시설과 호실을 선택하세요.');
    if (!groupName) return alert('그룹 명칭을 입력하세요.');
    if (droppedRoles.length === 0) return alert('포인트 롤을 추가하세요.');
    if (selectedGroupId !== 'new') return alert('수정 기능은 백엔드 구현 후 지원됩니다.');

    try {
      await apiService.createChartGroup({
        facility_id: selectedFacility,
        room_id: selectedRoom,
        group_name: groupName,
        role_ids: droppedRoles.map((r) => r.role_id),
      });
      alert('생성 완료!');
      handleRoomChange(selectedRoom);
    } catch (e) {
      alert('저장 실패');
    }
  };

  return (
    <div style={{ display: 'flex', flex: 1, minHeight: 0, height: '100%', gap: 10 }}>
      <LeftPanel
        facilities={facilities}
        rooms={rooms}
        selectedFacility={selectedFacility}
        selectedRoom={selectedRoom}
        room0Roles={room0Roles}
        targetRoomRoles={targetRoomRoles}
        droppedRoleIds={droppedRoles.map((r) => r.role_id)}
        onFacilityChange={handleFacilityChange}
        onRoomChange={handleRoomChange}
        onDragStart={onDragStart}
      />
      <RightPanel
        existingGroups={existingGroups}
        selectedGroupId={selectedGroupId}
        groupName={groupName}
        droppedRoles={droppedRoles}
        onGroupSelect={handleGroupSelect}
        onGroupNameChange={setGroupName}
        onDrop={onDrop}
        onDragOver={onDragOver}
        onRemoveRole={(id) => setDroppedRoles(droppedRoles.filter((r) => r.role_id !== id))}
        onReorderRoles={handleReorderRoles}
        onSaveGroup={handleSaveGroup}
      />
    </div>
  );
};

function useDebouncedCallback<T extends (...args: any[]) => void>(callback: T, delay: number) {
  const timeoutRef = useRef<number | null>(null);
  return useCallback((...args: Parameters<T>) => {
    if (timeoutRef.current) window.clearTimeout(timeoutRef.current);
    timeoutRef.current = window.setTimeout(() => callback(...args), delay);
  }, [callback, delay]);
}

export const ChartViewerPage: React.FC = () => {
  const [mode, setMode] = useState<ChartMode>('DEFAULT');
  const [scale, setScale] = useState<ChartScale>('1m');
  const [targetDate, setTargetDate] = useState<string>(getLocalDatetimeString(new Date()));
  const [refreshTrigger, setRefreshTrigger] = useState<number>(0); 
  
  const [facilities, setFacilities] = useState<any[]>([]);
  const [rooms, setRooms] = useState<any[]>([]);
  const [groups, setGroups] = useState<ChartGroup[]>([]);
  const [allRoles, setAllRoles] = useState<Role[]>([]);

  const [selectedFacility, setSelectedFacility] = useState<number | ''>('');
  const [selectedRoom, setSelectedRoom] = useState<number | ''>('');
  const [selectedGroupId, setSelectedGroupId] = useState<number | ''>('');
  const [checkedRoles, setCheckedRoles] = useState<number[]>([]);
  
  const [chartDataSets, setChartDataSets] = useState<any[]>([]);
  const [roleColors, setRoleColors] = useState<Record<number, string>>({});

  const [showMainInfo, setShowMainInfo] = useState<boolean>(true);
  const [showMainGrid, setShowMainGrid] = useState<boolean>(true);
  const [showSubInfo, setShowSubInfo] = useState<boolean>(false); 
  const [showSubGrid, setShowSubGrid] = useState<boolean>(true);

  const [openMenuRoleId, setOpenMenuRoleId] = useState<number | null>(null);
  const [activeGuidelines, setActiveGuidelines] = useState<number[]>([]);
  const [invertedBools, setInvertedBools] = useState<number[]>([]);
  const [isChartLoading, setIsChartLoading] = useState<boolean>(false); 

  const [posts, setPosts] = useState<any[]>([]);
  const [alarms, setAlarms] = useState<any[]>([]);
  const [hiddenPosts, setHiddenPosts] = useState<any[]>([]);
  const [selectedPostViewerIds, setSelectedPostViewerIds] = useState<number[]>([]);

  const [omniboxInput, setOmniboxInput] = useState('');
  const [omniboxSuggestions, setOmniboxSuggestions] = useState<any[]>([]);

  const mainChartRef = useRef<ReactECharts | null>(null);
  const subChartRef = useRef<ReactECharts | null>(null); 
  const loadedRangeRef = useRef<{ start: number, end: number } | null>(null);
  const [loadedRangeState, setLoadedRangeState] = useState<{ start: number, end: number } | null>(null);
  const [currentViewStart, setCurrentViewStart] = useState<number | null>(null);

  useEffect(() => { apiService.getFacilities().then(res => setFacilities(res.facilities || [])); }, []); 

  useEffect(() => {
    if (selectedFacility) {
      apiService.getRooms(Number(selectedFacility)).then(res => {
        const fetchedRooms = res.rooms || [];
        setRooms(fetchedRooms);
        if (fetchedRooms.length > 0) setSelectedRoom(fetchedRooms[0].id); 
      });
      setSelectedRoom(''); setGroups([]); setAllRoles([]); setSelectedGroupId(''); setCheckedRoles([]); 
      setChartDataSets([]); setPosts([]); setHiddenPosts([]); setAlarms([]); setSelectedPostViewerIds([]); setInvertedBools([]);
      loadedRangeRef.current = null; setLoadedRangeState(null); setCurrentViewStart(null);
    }
  }, [selectedFacility]); 

  useEffect(() => {
    if (selectedFacility && selectedRoom) {
      apiService.getChartGroups(Number(selectedFacility), Number(selectedRoom)).then(res => {
        const fetchedGroups = res.groups || [];
        setGroups(fetchedGroups);
        if (fetchedGroups.length > 0) setSelectedGroupId(fetchedGroups[0].group_id); 
      }); 
      apiService.getPointRoles(Number(selectedFacility), Number(selectedRoom)).then(res => {
        const roles = res.roles || [];
        setAllRoles(roles);
        const newColorMap: Record<number, string> = {};
        roles.forEach((r: Role, idx: number) => { newColorMap[r.role_id] = COLOR_PALETTE[idx % COLOR_PALETTE.length]; });
        setRoleColors(newColorMap);
      });
      setSelectedGroupId(''); setCheckedRoles([]); setChartDataSets([]); setPosts([]); setHiddenPosts([]); setAlarms([]); setSelectedPostViewerIds([]); setInvertedBools([]);
      loadedRangeRef.current = null; setLoadedRangeState(null); setCurrentViewStart(null);
    }
  }, [selectedRoom, selectedFacility]); 

  useEffect(() => {
    if (selectedGroupId) {
      const group = groups.find(g => g.group_id === Number(selectedGroupId));
      if (group && group.role_ids) setCheckedRoles(group.role_ids.filter(id => id !== null));
    } else {
      setCheckedRoles([]); setChartDataSets([]); setPosts([]); setHiddenPosts([]); setAlarms([]); setSelectedPostViewerIds([]); setInvertedBools([]);
      loadedRangeRef.current = null; setLoadedRangeState(null); setCurrentViewStart(null);
    }
  }, [selectedGroupId, groups]); 

  useEffect(() => {
    const q = omniboxInput.replace(/\s+/g, '').toLowerCase();
    if (!q) { setOmniboxSuggestions([]); return; }
    const sugs: any[] = [];
    facilities.forEach(f => { if (f.name.replace(/\s+/g, '').toLowerCase().includes(q)) sugs.push({ type: 'facility', id: f.id, name: f.name, label: `${f.name} (시설)` }); });
    if (selectedFacility) {
      rooms.forEach(r => { if (r.name.replace(/\s+/g, '').toLowerCase().includes(q)) sugs.push({ type: 'room', id: r.id, name: r.name, label: `${r.name} (호실)` }); });
    }
    if (selectedRoom) {
      groups.forEach(g => { if (g.group_name.replace(/\s+/g, '').toLowerCase().includes(q)) sugs.push({ type: 'group', id: g.group_id, name: g.group_name, label: `${g.group_name} (맵핑)` }); });
    }
    sugs.sort((a, b) => a.name.length - b.name.length);
    setOmniboxSuggestions(sugs);
  }, [omniboxInput, facilities, rooms, groups, selectedFacility, selectedRoom]);

  const handleOmniboxSelect = (sug: any) => {
    if (sug.type === 'facility') setSelectedFacility(sug.id);
    else if (sug.type === 'room') setSelectedRoom(sug.id);
    else if (sug.type === 'group') setSelectedGroupId(sug.id);
    setOmniboxInput('');
  };

  const handleOmniboxSubmit = () => {
    let remain = omniboxInput;
    let updatedDate = new Date(targetDate);
    let dateChanged = false; let timeChanged = false;

    const dateRegex1 = /(?:^|\s)(\d{4})[./-](\d{1,2})[./-](\d{1,2})(?:\s|$)/; 
    const dateRegex2 = /(?:^|\s)(\d{1,2})[./-](\d{1,2})(?:\s|$)/;             
    const dateRegex3 = /(?:^|\s)(\d{1,2})\s*월\s*(\d{1,2})\s*일(?:\s|$)/;     

    let m;
    if ((m = remain.match(dateRegex1))) { updatedDate.setFullYear(parseInt(m[1]), parseInt(m[2]) - 1, parseInt(m[3])); remain = remain.replace(m[0], ' '); dateChanged = true; } 
    else if ((m = remain.match(dateRegex3))) { updatedDate.setMonth(parseInt(m[1]) - 1, parseInt(m[2])); remain = remain.replace(m[0], ' '); dateChanged = true; } 
    else if ((m = remain.match(dateRegex2))) { updatedDate.setMonth(parseInt(m[1]) - 1, parseInt(m[2])); remain = remain.replace(m[0], ' '); dateChanged = true; }

    const timeRegex1 = /(?:^|\s)([01]\d|2[0-3]):([0-5]\d)(?:\s|$)/; 
    const timeRegex2 = /(?:^|\s)([01]\d|2[0-3])([0-5]\d)(?:\s|$)/;  
    
    if ((m = remain.match(timeRegex1))) { updatedDate.setHours(parseInt(m[1]), parseInt(m[2]), 0, 0); remain = remain.replace(m[0], ' '); timeChanged = true; } 
    else if ((m = remain.match(timeRegex2))) { updatedDate.setHours(parseInt(m[1]), parseInt(m[2]), 0, 0); remain = remain.replace(m[0], ' '); timeChanged = true; }

    if (dateChanged && !timeChanged) updatedDate.setHours(0, 0, 0, 0);
    if (dateChanged || timeChanged) setTargetDate(getLocalDatetimeString(updatedDate));

    const q = remain.replace(/\s+/g, '');
    if (q && omniboxSuggestions.length > 0) handleOmniboxSelect(omniboxSuggestions[0]);
    else setOmniboxInput('');
  };

  const fetchAndMergeChunk = async (fetchStartMs: number, fetchEndMs: number, direction: 'replace' | 'append' | 'prepend') => {
    if (!selectedFacility || !selectedRoom || checkedRoles.length === 0) return;
    setIsChartLoading(true);
    try {
      const startIso = new Date(fetchStartMs).toISOString();
      const endIso = new Date(fetchEndMs).toISOString();
      const res = await apiService.getHistoricalData(Number(selectedFacility), Number(selectedRoom), selectedGroupId, scale, startIso, endIso, checkedRoles);
      
      const newDatasets = res.data?.datasets || [];
      const newPosts = res.data?.posts || [];
      const newAlarms = res.data?.alarms || [];
      const newHiddenPosts = res.data?.hidden_posts || [];

      if (direction === 'replace') {
        setChartDataSets(newDatasets); setPosts(newPosts); setAlarms(newAlarms); setHiddenPosts(newHiddenPosts);
        loadedRangeRef.current = { start: fetchStartMs, end: fetchEndMs };
        setCurrentViewStart(fetchStartMs);
      } else {
        setChartDataSets(prev => prev.map(prevDs => {
          const newDs = newDatasets.find((n: any) => n.role_id === prevDs.role_id);
          if (!newDs) return prevDs;
          let mergedData = direction === 'append' ? [...prevDs.data, ...newDs.data] : [...newDs.data, ...prevDs.data];
          mergedData.sort((a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime());
          if (mergedData.length > 5000) mergedData = direction === 'append' ? mergedData.slice(-5000) : mergedData.slice(0, 5000);
          return { ...prevDs, data: mergedData };
        }));

        setPosts(prev => { const combined = direction === 'append' ? [...prev, ...newPosts] : [...newPosts, ...prev]; return Array.from(new Map(combined.map(p => [p.id, p])).values()); });
        setAlarms(prev => { const combined = direction === 'append' ? [...prev, ...newAlarms] : [...newAlarms, ...prev]; return Array.from(new Map(combined.map(a => [a.alarm_id, a])).values()); });

        if (loadedRangeRef.current) {
          const config = CHART_SCOPING_CONFIG[mode][scale];
          const maxMsRange = config.chunkSize * config.maxChunks;
          let newStart = Math.min(loadedRangeRef.current.start, fetchStartMs);
          let newEnd = Math.max(loadedRangeRef.current.end, fetchEndMs);
          if (newEnd - newStart > maxMsRange) {
            if (direction === 'append') newStart = newEnd - maxMsRange;
            if (direction === 'prepend') newEnd = newStart + maxMsRange;
          }
          loadedRangeRef.current = { start: newStart, end: newEnd };
        }
      }
      setLoadedRangeState(loadedRangeRef.current);
    } catch (err) { console.error(err); } 
    finally { setIsChartLoading(false); }
  };

  useEffect(() => {
    if (selectedFacility && selectedRoom && checkedRoles.length > 0 && targetDate) {
      const endMs = new Date(targetDate).getTime();
      const chunkMs = CHART_SCOPING_CONFIG[mode][scale].chunkSize;
      fetchAndMergeChunk(endMs - chunkMs, endMs, 'replace');
    } else {
      setChartDataSets([]); setPosts([]); setAlarms([]); setHiddenPosts([]);
      loadedRangeRef.current = null; setLoadedRangeState(null); setCurrentViewStart(null);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [selectedFacility, selectedRoom, scale, targetDate, checkedRoles, refreshTrigger]); 

  const handleDataZoomFetch = useDebouncedCallback((viewStartMs: number, viewEndMs: number) => {
    if (isChartLoading || !loadedRangeRef.current || checkedRoles.length === 0) return;
    const config = CHART_SCOPING_CONFIG[mode][scale];
    const buffer = config.chunkSize * 0.2; 
    if (viewStartMs < loadedRangeRef.current.start + buffer) fetchAndMergeChunk(loadedRangeRef.current.start - config.chunkSize, loadedRangeRef.current.start, 'prepend');
    else if (viewEndMs > loadedRangeRef.current.end - buffer) fetchAndMergeChunk(loadedRangeRef.current.end, loadedRangeRef.current.end + config.chunkSize, 'append');
  }, 500);

  const handleZoomCapture = useCallback((start: number, end: number) => {
    setCurrentViewStart(start); 
    handleDataZoomFetch(start, end);
  }, [handleDataZoomFetch]);

  const isBoolRole = (roleId: number) => {
    const role = allRoles.find(r => String(r.role_id) === String(roleId));
    if (!role) return false;
    return role.data_type ? role.data_type === 'BOOL' : role.role_name.toLowerCase().includes('bool');
  }; 

  const handleModeToggle = () => {
    if (isChartLoading) return; 
    setMode(prev => {
      const nextMode = prev === 'DEFAULT' ? 'EXTENT' : 'DEFAULT';
      if (nextMode === 'EXTENT') {
        const boolRoles = checkedRoles.filter(id => isBoolRole(id));
        const mainRoles = checkedRoles.filter(id => !isBoolRole(id));
        if (mainRoles.length > 1) setCheckedRoles([...boolRoles, mainRoles[0]]);
      }
      return nextMode;
    });
  }; 

  const handleRoleToggle = (roleId: number) => {
    if (isChartLoading) return; 
    const isBool = isBoolRole(roleId);
    setCheckedRoles(prev => prev.includes(roleId) ? prev.filter(id => id !== roleId) : (mode === 'EXTENT' && !isBool ? [...prev.filter(id => isBoolRole(id)), roleId] : [...prev, roleId]));
  }; 

  const handleInvertToggle = (roleId: number) => setInvertedBools(prev => prev.includes(roleId) ? prev.filter(id => id !== roleId) : [...prev, roleId]);
  const handleColorChange = (roleId: number, color: string) => setRoleColors(prev => ({ ...prev, [roleId]: color }));
  const handleGuidelineToggle = (roleId: number) => setActiveGuidelines(prev => prev.includes(roleId) ? prev.filter(id => id !== roleId) : [...prev, roleId]);

  const syncChartDataZoom = (sourceRef: React.RefObject<ReactECharts | null>, targetRef: React.RefObject<ReactECharts | null>) => {
    const option = sourceRef.current?.getEchartsInstance().getOption() as any;
    if (option?.dataZoom?.length > 0) targetRef.current?.getEchartsInstance().dispatchAction({ type: 'dataZoom', startValue: option.dataZoom[0].startValue, endValue: option.dataZoom[0].endValue });
  }; 

  const applyOneHourView = () => {
    const option = mainChartRef.current?.getEchartsInstance().getOption() as any;
    let centerTime = new Date(targetDate).getTime();
    if (option?.dataZoom?.length > 0 && option.dataZoom[0].startValue) centerTime = (option.dataZoom[0].startValue + option.dataZoom[0].endValue) / 2;
    const zoomPayload = { type: 'dataZoom', startValue: centerTime - 1800000, endValue: centerTime + 1800000 };
    mainChartRef.current?.getEchartsInstance().dispatchAction(zoomPayload);
    subChartRef.current?.getEchartsInstance().dispatchAction(zoomPayload);
  }; 

  const handleQuickTimeSelect = (hours: number) => {
    if (isChartLoading) return; 
    const now = new Date(); setTargetDate(getLocalDatetimeString(now)); 
    setTimeout(() => {
      const zoomPayload = { type: 'dataZoom', startValue: now.getTime() - (hours * 3600000), endValue: now.getTime() };
      mainChartRef.current?.getEchartsInstance().dispatchAction(zoomPayload);
      subChartRef.current?.getEchartsInstance().dispatchAction(zoomPayload);
    }, 300); 
  }; 

  const handleQuickDaySelect = (targetDayOfWeek: number) => {
    if (isChartLoading) return; 
    const now = new Date();
    let diff = now.getDay() - targetDayOfWeek;
    if (diff < 0) diff += 7;
    setTargetDate(getLocalDatetimeString(new Date(now.getTime() - (diff * 86400000))));
  }; 

  const handleChartClick = (type: 'post' | 'alarm', ids: number | number[]) => {
    if (type === 'post') setSelectedPostViewerIds(Array.isArray(ids) ? ids : [ids]);
  };

  const handleHidePost = async (postId: number) => {
    if (!selectedGroupId) return;
    try { await apiService.hidePostFromGroup(Number(selectedGroupId), postId); setSelectedPostViewerIds(prev => prev.filter(id => id !== postId)); setRefreshTrigger(prev => prev + 1); } catch (e) {}
  };

  const handleRestorePost = async (postId: number) => {
    if (!selectedGroupId) return;
    try { await apiService.showPostInGroup(Number(selectedGroupId), postId); setRefreshTrigger(prev => prev + 1); } catch (e) {}
  };

  const handleCopySearchQuery = () => {
    if (!selectedFacility || !selectedRoom || !currentViewStart) return;
    const facName = facilities.find(f => f.id === selectedFacility)?.name || '';
    const roomName = rooms.find(r => r.id === selectedRoom)?.name || '';
    const d = new Date(currentViewStart);
    const query = `${facName} ${roomName} ${d.getMonth() + 1}/${d.getDate()} ${d.getHours().toString().padStart(2, '0')}:${d.getMinutes().toString().padStart(2, '0')}`;
    navigator.clipboard.writeText(query).then(() => alert(`클립보드에 복사되었습니다:\n${query}`));
  };

  const mainDataSets = chartDataSets.filter(ds => ds.role_name && !isBoolRole(ds.role_id));
  const subDataSets = chartDataSets.filter(ds => ds.role_name && isBoolRole(ds.role_id)); 

  // 💡 [UI 핵심 수정] 최상위 컨테이너에 overflowY: 'auto' 적용
  return (
    <div style={{ display: 'flex', flexDirection: 'column', height: '100%', overflowY: 'auto', gap: 10, boxSizing: 'border-box', paddingRight: 5 }}>
      
      {/* 상단: 차트 + 검색/스코핑 패널 (최소 높이 보장) */}
      <div style={{ display: 'flex', gap: 10, minHeight: 650, flexShrink: 0 }}>
        
        {/* 왼쪽 차트 영역 */}
        <div style={{ flex: 3, border: '2px solid #2196f3', display: 'flex', flexDirection: 'column', padding: 10, gap: 10 }}>
          <TopToolBar showInfo={showMainInfo} showGrid={showMainGrid} targetDate={targetDate} isChartLoading={isChartLoading} onCopy={handleCopySearchQuery} onToggleInfo={() => setShowMainInfo(!showMainInfo)} onToggleGrid={() => setShowMainGrid(!showMainGrid)} onOneHourView={applyOneHourView} onDateChange={setTargetDate} />
          <div style={{ border: '2px solid #ffffff', flex: 2, display: 'flex', flexDirection: 'column', backgroundColor: '#1e1e1e', minHeight: 0 }}>
            <MainChartViewer echartsRef={mainChartRef} datasets={mainDataSets} roleColors={roleColors} showInfo={showMainInfo} showGrid={showMainGrid} activeGuidelines={activeGuidelines} isLoading={isChartLoading} posts={posts} alarms={alarms} onChartClick={handleChartClick} loadedRange={loadedRangeState} onDataZoomPan={handleZoomCapture} />
          </div>
          <LowerToolBar showInfo={showSubInfo} showGrid={showSubGrid} onToggleInfo={() => setShowSubInfo(!showSubInfo)} onToggleGrid={() => setShowSubGrid(!showSubGrid)} onSyncTop={() => syncChartDataZoom(mainChartRef, subChartRef)} onSyncSub={() => syncChartDataZoom(subChartRef, mainChartRef)} />
          <div style={{ border: '2px solid #ffffff', flex: 1, display: 'flex', flexDirection: 'column', backgroundColor: '#1e1e1e', minHeight: 0 }}>
            <SubChartViewer echartsRef={subChartRef} datasets={subDataSets} roleColors={roleColors} showInfo={showSubInfo} showGrid={showSubGrid} isLoading={isChartLoading} posts={posts} alarms={alarms} invertedBools={invertedBools} loadedRange={loadedRangeState} />
          </div>
        </div>

        {/* 오른쪽 패널 영역 */}
        <div style={{ flex: 1, border: '2px solid #e91e63', display: 'flex', flexDirection: 'column', padding: 10, gap: 10 }}>
          <SmartOmniboxPanel input={omniboxInput} suggestions={omniboxSuggestions} isChartLoading={isChartLoading} onInputChange={setOmniboxInput} onSubmit={handleOmniboxSubmit} onSelect={handleOmniboxSelect} />
          <ScopingPanel mode={mode} scale={scale} facilities={facilities} rooms={rooms} selectedFacility={selectedFacility} selectedRoom={selectedRoom} isChartLoading={isChartLoading} onModeToggle={handleModeToggle} onScaleChange={setScale} onFacilityChange={setSelectedFacility} onRoomChange={setSelectedRoom} onQuickTime={handleQuickTimeSelect} onQuickDay={handleQuickDaySelect} />
          <PostViewerPanel selectedPostIds={selectedPostViewerIds} posts={posts} selectedGroupId={selectedGroupId} onHidePost={handleHidePost} />
        </div>
      </div>

      {/* 하단: 맵핑 활성화 + 블랙리스트 (내용에 따라 자동 확장) */}
      <div style={{ display: 'flex', gap: 10, flexShrink: 0 }}>
        <MappingActivePanel groups={groups} allRoles={allRoles} roleColors={roleColors} selectedGroupId={selectedGroupId} selectedRoom={selectedRoom} isChartLoading={isChartLoading} checkedRoles={checkedRoles} activeGuidelines={activeGuidelines} invertedBools={invertedBools} openMenuRoleId={openMenuRoleId} onGroupChange={setSelectedGroupId} onRoleToggle={handleRoleToggle} onMenuToggle={setOpenMenuRoleId} onColorChange={handleColorChange} onGuidelineToggle={handleGuidelineToggle} onInvertToggle={handleInvertToggle} isBoolRole={isBoolRole} />
        <BlacklistPanel hiddenPosts={hiddenPosts} onRestore={handleRestorePost} />
      </div>

    </div>
  );
};
