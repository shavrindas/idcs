from firebase_admin import credentials
from src.module.module_router import api_router
from src.module.FServer.controllers.connection_manager import fserver_conn_manager

@asynccontextmanager
async def lifespan(app: FastAPI):
    print("LOG[STARTUP] : IDCool Server is starting up...")
    
    bg_task = None
    if os.getenv("ENABLE_FACILITY_SERVER", "false").lower() == "true":
        bg_task = asyncio.create_task(fserver_conn_manager.maintain_connection())
    yield

    
    print("LOG[SHUTDOWN] : Shutting down IDCool Server...")
    if bg_task:
        bg_task.cancel()
    await engine.dispose()

    title="IDCool Systems API",
    description="IDCool Systems 서버 백엔드 API",
    version="2.0.0",
    lifespan=lifespan
)

    CORSMiddleware,
    allow_origins=["*"], # 개발 단계이므로 전체 허용

    allow_methods=["*"],
    allow_headers=["*"]
)

def root():
    return {"message": "Server is running"}

import os
import asyncio
from firebase_admin import messaging

async def send_multicast_fcm(tokens: list[str], title: str, body: str):
    if not tokens:
        return

    message = messaging.MulticastMessage(
        data={
            "title": str(title),
            "body": str(body),
        },
        tokens=tokens,
        android=messaging.AndroidConfig(
            priority='high',
        ),
    )
    
    response = await asyncio.to_thread(messaging.send_each_for_multicast, message)
    print(f"[FCM 다중 전송] 성공: {response.success_count}건, 실패: {response.failure_count}건")

import enum

from sqlalchemy.orm import relationship
from src.config.database import Base


class ChatRoomType(str, enum.Enum):
    DIRECT = "DIRECT"
    GROUP = "GROUP"
    FACILITY = "FACILITY"


class MessageType(str, enum.Enum):
    TEXT = "TEXT"
    IMAGE = "IMAGE"
    FILE = "FILE"


class ChatRoom(Base):
    __tablename__ = "chatrooms"

    room_id = Column(Integer, primary_key=True, autoincrement=True)
    facility_id = Column(Integer, ForeignKey("facilities_table.id", ondelete="CASCADE"), nullable=True)

    unread_count = Column(Integer, nullable=False, default=0)

    # 관계 정의

        UniqueConstraint("room_id", "user_id", name="uq_room_user"),

import jwt
import os
import secrets
import hashlib
from datetime import datetime, timedelta, timezone

API_KEY_PREFIX = "idk_live_"

# 운영 환경 변수에서 로드 (미설정 시 즉시 에러 발생)
SYSTEM_SECRET_KEY = os.getenv("SYSTEM_SECRET_KEY")
if not SYSTEM_SECRET_KEY:
    # 테스트/개발 환경 편의를 위한 경고 겸 기본 fallback을 두거나, 엄격하게 예외 처리
    # raise RuntimeError("CRITICAL: SYSTEM_SECRET_KEY 환경 변수가 설정되지 않았습니다.")
    SYSTEM_SECRET_KEY = "idcool_system_super_secret_key_v2_secure_32bytes_min"


ALGORITHM = "HS256"


def create_system_token(node_id: int, facility_id: int, expires_delta: timedelta = timedelta(hours=6)):
    """시스템(FServer) 접속용 JWT 발급"""
    expire = datetime.now(timezone.utc) + expires_delta
    payload = {
        "sub": str(node_id),
        "facility_id": facility_id,
        "type": "system",
        "exp": expire
    }
    return jwt.encode(payload, SYSTEM_SECRET_KEY, algorithm=ALGORITHM)

def verify_system_token(token: str):
    """X-System-Token 검증 로직"""
    try:
        payload = jwt.decode(token, SYSTEM_SECRET_KEY, algorithms=[ALGORITHM])
        if payload.get("type") != "system":
            return None
        return payload
    except jwt.ExpiredSignatureError:
        return "EXPIRED"
    except jwt.InvalidTokenError:
        return None

def generate_facility_api_key() -> tuple[str, str]:
    """(원문 API Key, DB 저장용 해시값) 생성"""
    raw_secret = secrets.token_urlsafe(32)
    raw_key = f"{API_KEY_PREFIX}{raw_secret}"
    key_hash = hash_api_key(raw_key)
    return raw_key, key_hash

def hash_api_key(raw_key: str) -> str:
    """API Key SHA-256 해싱"""
    return hashlib.sha256(raw_key.encode()).hexdigest()

from fastapi import APIRouter, Depends, HTTPException, WebSocket, WebSocketDisconnect
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.future import select
from sqlalchemy import func
from pydantic import BaseModel

from src.config.database import get_db, AsyncSessionLocal
from src.module.Core.models.Facility import FacilityNode
from src.module.Core.security import create_system_token, verify_system_token, hash_api_key

router = APIRouter(prefix="/sys-auth", tags=["System Authentication"])
stream_router = APIRouter(tags=["System Stream"])

class SystemLoginReq(BaseModel):
    node_code: str
    api_key: str

@router.post("/login")
async def system_login(req: SystemLoginReq, db: AsyncSession = Depends(get_db)):
    result = await db.execute(select(FacilityNode).where(FacilityNode.node_code == req.node_code))
    node = result.scalars().first()
    
    if not node or not node.is_active:
        raise HTTPException(status_code=401, detail="유효하지 않은 시설 노드입니다.")
    
    if node.api_key_hash != hash_api_key(req.api_key):
        raise HTTPException(status_code=401, detail="API Key가 일치하지 않습니다.")

    token = create_system_token(node_id=node.id, facility_id=node.facility_id)
    
    node.status = "ONLINE"
    node.last_heartbeat_at = func.now()
    await db.commit()
    
    return {
        "access_token": token, 
        "token_type": "bearer", 
        "facility_id": node.facility_id,
        "node_id": node.id
    }

@stream_router.websocket("/data-stream/{facility_id}")
async def facility_data_stream(websocket: WebSocket, facility_id: int):
    token = websocket.headers.get("X-System-Token") or websocket.query_params.get("token")
    await websocket.accept()

    if not token:
        await websocket.close(code=4001, reason="토큰이 없습니다.")
        return

    payload = verify_system_token(token)
    if payload == "EXPIRED" or not payload or int(payload.get("facility_id", -1)) != int(facility_id):
        await websocket.close(code=4001, reason="유효하지 않거나 만료된 시스템 토큰입니다.")
        return

    node_id = int(payload.get("sub"))
    print(f"[CServer] 시설({facility_id}) 시스템 노드(ID: {node_id}) 연결 성공 및 스트리밍 시작")
    
    try:

import asyncio
import aiohttp
import websockets
import json
import os

CENTER_API_URL = os.getenv("CENTER_API_URL", "http://127.0.0.1:8900/api/v1/cserver/sys-auth/login")
CENTER_WS_URL = os.getenv("CENTER_WS_URL", "ws://127.0.0.1:8900/api/v1/cserver/data-stream")
NODE_CODE = os.getenv("NODE_CODE", "NODE_001")
API_KEY = os.getenv("API_KEY", "dummy_api_key_123")

class FacilityConnectionManager:
    def __init__(self):
        self.token = None
        self.facility_id = None
        self.is_connected = False
        self.active_ws = None
        
    async def authenticate(self):
        async with aiohttp.ClientSession() as session:
            try:
                print(f"[FServer] 중앙 계정 시스템 인증 시도... ({NODE_CODE})")
                async with session.post(CENTER_API_URL, json={
                    "node_code": NODE_CODE,
                    "api_key": API_KEY
                }) as resp:
                    if resp.status == 200:
                        data = await resp.json()
                        self.token = data["access_token"]
                        self.facility_id = data["facility_id"]
                        print(f"[FServer] 시스템 인증 완료! 위임 토큰 발급됨 (Facility ID: {self.facility_id})")
                        return True
                    else:
                        print(f"[FServer] 인증 실패: HTTP {resp.status}")
                        return False
            except Exception as e:
                print(f"[FServer] 인증 서버 접근 불가: {e}")
                return False

    async def maintain_connection(self):
        while True:
            if not self.token:
                success = await self.authenticate()
                if not success:
                    await asyncio.sleep(3)
                    continue

            headers = {"X-System-Token": self.token}
            ws_url = f"{CENTER_WS_URL}/{self.facility_id}?token={self.token}"
            
            try:
                print(f"🔄 [FServer] 중앙 데이터 스트리밍 연결 시도 중... ({ws_url})")
                try:
                    ws_ctx = websockets.connect(ws_url, additional_headers=headers)
                except TypeError:
                    ws_ctx = websockets.connect(ws_url, extra_headers=headers)

                async with ws_ctx as ws:
                    self.active_ws = ws
                    self.is_connected = True
                    print(f"[FServer] 중앙 시스템과 데이터 연결 확립 (Facility ID: {self.facility_id})")
                    
                    while True:
                        dummy_payload = {"status": "OK", "temp_pv": 24.5} 
                        await ws.send(json.dumps(dummy_payload))
                        await asyncio.sleep(1)
                        
            except websockets.exceptions.ConnectionClosed as e:
                self.is_connected = False
                self.active_ws = None
                print(f"[FServer] 통신 단절 (Code: {e.code}, Reason: {e.reason})")
                if e.code == 4001:
                    print("[FServer] 시스템 토큰 만료됨. 재인증을 요청합니다.")
                    self.token = None 
            except asyncio.CancelledError:
                # Lifespan 종료 시 호출
                await self.graceful_shutdown()
                break
            except Exception as e:
                self.is_connected = False
                self.active_ws = None
                print(f"[FServer] 예상치 못한 네트워크 에러: {e}")
            
            await asyncio.sleep(3)

    async def graceful_shutdown(self):
        """프로세스 종료 시 중앙에 정상 종료를 알림"""
        if self.active_ws and not self.active_ws.closed:
            try:
                print("[FServer] 중앙 서버에 정상 종료 통보 전송 중...")
                await self.active_ws.send("__SHUTDOWN__")
                await self.active_ws.close()
            except Exception:
                pass
        self.is_connected = False

    db.add(new_token)

@router.get("/integrated")
async def get_integrated_posts(
    user_id: int, 
    page: int = Query(1, ge=1), 
    limit: int = Query(30, le=100),
    search_type: str = Query("all"), # all, title, creator, content
    keyword: Optional[str] = Query(None),
    db: AsyncSession = Depends(get_db)
):
    user_query = text("SELECT role FROM user_table WHERE id = :user_id")
    user_res = await db.execute(user_query, {"user_id": user_id})
    user = user_res.fetchone()

    if not user or user.role not in ['ADMIN', 'DEVELOPER']:
        raise HTTPException(status_code=403, detail="통합 조회 보드에 접근할 권한이 없습니다.")

    offset = (page - 1) * limit

    # 검색 쿼리 조건 생성
    where_clause = "WHERE 1=1"
    params = {"limit": limit, "offset": offset, "user_id": user_id}

    if keyword and keyword.strip():
        params["keyword"] = f"%{keyword.strip()}%"
        if search_type == "title":
            where_clause += " AND p.title ILIKE :keyword"
        elif search_type == "creator":

    check_query = text("SELECT creator_id FROM posts WHERE post_id = :post_id")

    update_query = text("""
        UPDATE posts 
        SET title = COALESCE(:title, title),
            content = COALESCE(:content, content),
            updater_id = :user_id
        WHERE post_id = :post_id
    """)
    await db.execute(update_query, {
        "title": req.title, 
        "content": req.content, 
        "user_id": user_id, 
        "post_id": post_id
    })

    # 3. 추가된 이미지가 있다면 맵핑
    if req.image_ids:
        mapping_query = text("""
            UPDATE post_images 
            SET post_id = :post_id 
            WHERE image_id = ANY(:image_ids)
        """)
        await db.execute(mapping_query, {"post_id": post_id, "image_ids": req.image_ids})

    await db.commit()
    return {"status": "success", "message": "포스트 수정 완료"}

@router.get("/compare/{facility_id}/available-roles")
async def get_comparable_roles(facility_id: int, db: AsyncSession = Depends(get_db)):
    """
    같은 시설 안에서 2개 이상의 호실에 공통으로 존재하는 role_name 목록을 반환한다.
    (예: 모든 냉동실에 있는 "고내온도" 센서를 한 번에 비교하기 위한 선택지)
    """
    # bool_ 타입은 꺾은선 그래프로 겹쳐 비교하기 적합하지 않아 목록에서 제외한다
    # (프론트의 MainChartViewer는 수치형 mean 값을 기준으로 그리기 때문).
    query = text("""
        SELECT pr.role_name, COUNT(DISTINCT pr.room_id) AS room_count,
               MAX(prm.display_name) AS display_name
        FROM point_role_table pr
        LEFT JOIN point_role_metadata prm ON pr.id = prm.role_id
        WHERE pr.facility_id = :facility_id AND pr.is_current = TRUE
          AND pr.role_name NOT ILIKE 'bool\\_%' ESCAPE '\\'
        GROUP BY pr.role_name
        HAVING COUNT(DISTINCT pr.room_id) > 1
        ORDER BY pr.role_name
    """)
    rows = await db.execute(query, {"facility_id": facility_id})
    roles = [
        {
            "role_name": row.role_name,
            "display_name": row.display_name or row.role_name,
            "room_count": row.room_count,
        }
        for row in rows.fetchall()
    ]
    return {"status": "success", "data": {"roles": roles}}


@router.get("/compare/{facility_id}")
async def get_compare_data(
    facility_id: int,
    role_name: str = Query(..., description="호실 간 비교할 role_name"),
    scale: str = Query("1m", description="데이터 해상도"),
    time_range: str = Query("1d", description="조회 기간"),
    db: AsyncSession = Depends(get_db)
):
    """
    같은 시설의 여러 호실에서 동일한 role_name(예: 고내온도)을 가진 값들을
    호실별로 한 그래프에 겹쳐볼 수 있도록 반환한다.
    """
    if scale not in SCALE_MAPPING or time_range not in TIME_RANGE_MAPPING:
        raise HTTPException(status_code=400, detail="유효하지 않은 파라미터입니다.")

    bucket_interval = SCALE_MAPPING[scale]
    range_interval = TIME_RANGE_MAPPING[time_range]

    roles_query = text("""
        SELECT pr.id AS role_id, pr.room_id, fr.name AS room_name
        FROM point_role_table pr
        JOIN facilities_room_table fr ON fr.id = pr.room_id
        WHERE pr.facility_id = :facility_id AND pr.role_name = :role_name AND pr.is_current = TRUE
        ORDER BY fr.name
    """)
    role_rows = (await db.execute(roles_query, {"facility_id": facility_id, "role_name": role_name})).fetchall()

    if not role_rows:
        return {"status": "success", "data": {"datasets": []}, "message": "비교할 호실 데이터가 없습니다."}

    target_roles = [row.role_id for row in role_rows]
    room_by_role = {row.role_id: {"room_id": row.room_id, "room_name": row.room_name} for row in role_rows}

    r_name_lower = role_name.lower()
    if "temp_" in r_name_lower:
        table_name, agg_columns, data_key_mapping = "temperature_numerical_logs", "AVG(temp_mean) AS mean, MIN(temp_min) AS min, MAX(temp_max) AS max", {"mean": "mean", "min": "min", "max": "max"}
    elif "gear_" in r_name_lower:
        table_name, agg_columns, data_key_mapping = "gear_numerical_logs", "AVG(gear_mean) AS mean, MIN(gear_min) AS min, MAX(gear_max) AS max", {"mean": "mean", "min": "min", "max": "max"}
    elif "static_" in r_name_lower:
        table_name, agg_columns, data_key_mapping = "equipment_static_logs", "AVG(setnum) AS mean", {"mean": "mean"}
    elif "bool_" in r_name_lower:
        table_name, agg_columns, data_key_mapping = "bool_logs", "BOOL_OR(status_value) AS status", {"status": "status"}
    else:
        return {"status": "success", "data": {"datasets": []}, "message": "지원되지 않는 데이터 타입입니다."}

    max_time_query = text(f"SELECT MAX(timestamp) FROM {table_name} WHERE role_id = ANY(:role_ids)")
    latest_time = (await db.execute(max_time_query, {"role_ids": target_roles})).scalar()

    if not latest_time:
        return {"status": "success", "data": {"datasets": []}, "message": "기록된 로그가 없습니다."}

    datasets_map: Dict[int, Any] = {
        r_id: {
            "role_id": r_id,
            "room_id": room_by_role[r_id]["room_id"],
            "room_name": room_by_role[r_id]["room_name"],
            "role_name": role_name,
            "data": [],
        } for r_id in target_roles
    }

    query = text(f"""
        SELECT role_id, time_bucket('{bucket_interval}', timestamp) AS bucket, {agg_columns}
        FROM {table_name}
        WHERE role_id = ANY(:role_ids)
          AND timestamp >= (CAST(:latest_time AS timestamptz) - INTERVAL '{range_interval}')
          AND timestamp <= CAST(:latest_time AS timestamptz)
        GROUP BY role_id, bucket ORDER BY bucket ASC
    """)
    rows = await db.execute(query, {"role_ids": target_roles, "latest_time": latest_time})
    for r in rows.fetchall():
        r_dict = dict(r._mapping)
        point = {"timestamp": r_dict["bucket"].isoformat()}
        for db_key, out_key in data_key_mapping.items(): point[out_key] = r_dict[db_key]
        datasets_map[r_dict["role_id"]]["data"].append(point)

    return {
        "status": "success",
        "meta": {"facility_id": facility_id, "role_name": role_name, "scale": scale, "time_range": time_range, "reference_time": latest_time.isoformat()},
        "data": {"datasets": list(datasets_map.values())}
    }


@router.get("/{facility_id}/{room_id}")
async def get_chart_data(
    facility_id: int,
    room_id: int,
    scale: str = Query("1m", description="데이터 해상도"),
    time_range: str = Query("1d", description="조회 기간"),
    role_ids: str = Query(..., description="조회할 PointRole ID 목록 (콤마로 구분)"),

        raise HTTPException(status_code=400, detail="유효하지 않은 파라미터입니다.")
    
    bucket_interval = SCALE_MAPPING[scale]
    range_interval = TIME_RANGE_MAPPING[time_range]

    # 1. PointRole 메타데이터 조회 (타입 구분을 위해 role_name 활용)
    # [주석] chart_group_table 의 구조와 그룹핑은 타입을 가리지 않습니다.
    # 포인트롤 자체가 곧 타입 정보를 가지므로(role_name), 백엔드에서 이를 분석해 각각의 하이퍼테이블을 조회합니다.

    # 역할별 쿼리 대상 분류
    temp_roles, gear_roles, static_roles, bool_roles = [], [], [], []
    for r_id, r_info in roles_info.items():
        r_name = r_info["role_name"].lower()
        if "temp_" in r_name: temp_roles.append(r_id)
        elif "gear_" in r_name: gear_roles.append(r_id)
        elif "static_" in r_name: static_roles.append(r_id)
        elif "bool_" in r_name: bool_roles.append(r_id)

    # 2. 통합 타임라인을 위한 기준 시간(가장 최근 데이터 시간) 확보
    # 전체 쿼리를 단순화하기 위해 temperature 테이블 기준으로 최근 시간을 잡음 (실 환경에선 전체 테이블 MAX 비교 권장)
    max_time_query = text("SELECT MAX(timestamp) FROM temperature_numerical_logs WHERE role_id = ANY(:role_ids)")
    latest_time = (await db.execute(max_time_query, {"role_ids": temp_roles if temp_roles else target_roles})).scalar()

    if not latest_time:
        return {"status": "success", "data": {"datasets": []}, "message": "기록된 로그가 없습니다."}

    # 3. 타입별 데이터 집계 및 병합 헬퍼 함수
    datasets_map: Dict[int, Any] = {
        r_id: {
            "role_id": r_id, "role_name": roles_info[r_id]["role_name"],
            "target_value": roles_info[r_id]["target_value"],
            "target_low_limit": roles_info[r_id]["target_low_limit"],
            "target_high_limit": roles_info[r_id]["target_high_limit"],
            "data": []
        } for r_id in roles_info.keys()
    }

    async def fetch_and_map(table_name: str, agg_columns: str, type_roles: List[int], data_key_mapping: dict):
        if not type_roles: return
        query = text(f"""
            SELECT role_id, time_bucket('{bucket_interval}', timestamp) AS bucket, {agg_columns}
            FROM {table_name}
            WHERE role_id = ANY(:role_ids)
              AND timestamp >= (CAST(:latest_time AS timestamptz) - INTERVAL '{range_interval}')
              AND timestamp <= CAST(:latest_time AS timestamptz)
            GROUP BY role_id, bucket ORDER BY bucket ASC
        """)
        rows = await db.execute(query, {"role_ids": type_roles, "latest_time": latest_time})
        for r in rows.fetchall():
            r_dict = dict(r._mapping)
            point = {"timestamp": r_dict["bucket"].isoformat()}
            for db_key, out_key in data_key_mapping.items(): point[out_key] = r_dict[db_key]
            datasets_map[r_dict["role_id"]]["data"].append(point)

    # 4. 각 하이퍼테이블 병렬/순차 조회
    await fetch_and_map("temperature_numerical_logs", "AVG(temp_mean) AS mean, MIN(temp_min) AS min, MAX(temp_max) AS max", temp_roles, {"mean": "mean", "min": "min", "max": "max"})
    await fetch_and_map("gear_numerical_logs", "AVG(gear_mean) AS mean, MIN(gear_min) AS min, MAX(gear_max) AS max", gear_roles, {"mean": "mean", "min": "min", "max": "max"})
    await fetch_and_map("equipment_static_logs", "AVG(setnum) AS mean", static_roles, {"mean": "mean"})
    await fetch_and_map("bool_logs", "BOOL_OR(status_value) AS status", bool_roles, {"status": "status"})

    # 5. 조회 중인 포인트롤에 걸린 알람을 같은 기간으로 함께 조회 (그래프 위에 발생 시점 표시용)

          AND timestamp <= CAST(:latest_time AS timestamptz)
        ORDER BY timestamp ASC
    """.format(range_interval=range_interval))
    alarm_rows = await db.execute(alarm_query, {"role_ids": target_roles, "latest_time": latest_time})
    alarms = [
        {
            "alarm_id": row.alarm_id,
            "role_id": row.role_id,
            "alarm_name": row.alarm_name,
            "severity": row.severity,
            "timestamp": row.timestamp.isoformat(),
        }
        for row in alarm_rows.fetchall()
    ]

    return {
        "status": "success",
        "meta": {"facility_id": facility_id, "room_id": room_id, "scale": scale, "time_range": time_range, "reference_time": latest_time.isoformat()},
        "data": {"datasets": list(datasets_map.values()), "alarms": alarms}

class ConnectionManager:
    def __init__(self):
        self.active_connections: Dict[int, List[WebSocket]] = {}

    async def connect(self, websocket: WebSocket, room_id: int):
        await websocket.accept()
        if room_id not in self.active_connections:
            self.active_connections[room_id] = []
        self.active_connections[room_id].append(websocket)

    def disconnect(self, websocket: WebSocket, room_id: int):
        if room_id in self.active_connections:
            self.active_connections[room_id].remove(websocket)
            if not self.active_connections[room_id]:
                del self.active_connections[room_id]

    async def broadcast_to_room(self, room_id: int, message: dict):
        if room_id in self.active_connections:
            for connection in self.active_connections[room_id]:
                await connection.send_text(json.dumps(message))

        FROM chatrooms cr
        JOIN chatparticipants cp ON cr.room_id = cp.room_id

                if r.other_user_role in ['ADMIN', 'DEVELOPER']: priority = 2

        if r.type == 'DIRECT' and not room_name_display:

class RegistTokenRequest(BaseModel):
    fcm_token: str
    user_id: int

@router.post('/register')
async def registToken(user_in: RegistTokenRequest, db: AsyncSession = Depends(get_db)):
    """Box #1: FCM 토큰 등록 (비동기 처리)"""
    
    # 1. 중복 등록 확인
    mapping_result = await db.execute(
        select(UserFcmToken).where(UserFcmToken.fcm_token == user_in.fcm_token)
    )
    existing_token = mapping_result.scalars().first()

    # 이미 동일한 토큰이 등록된 경우
    if existing_token:
        # 소유 유저가 바뀌었을 가능성이 있으므로 user_id 업데이트 후 저장
        if existing_token.user_id != user_in.user_id:
            existing_token.user_id = user_in.user_id
            await db.commit()
            return {"message": "기존 토큰의 사용자 정보가 갱신되었습니다.", "user_id": user_in.user_id}
        
        return {"message": "이미 등록된 토큰입니다.", "user_id": user_in.user_id}

    new_token = UserFcmToken(
        fcm_token=user_in.fcm_token, 
        user_id=user_in.user_id
    )
    
    db.add(new_token)
    await db.commit()

    return {"message": "토큰이 성공적으로 등록되었습니다.", "user_id": new_token.user_id}



@router.get("/{user_id}")
async def get_alarms_for_user(user_id: int, db: AsyncSession = Depends(get_db)):
    stmt = (
        select(
            Alarm, 
            PointRole.facility_id, 
            Facility.name.label("facility_name"), 
            FacilityRoom.id.label("room_id"), 
            FacilityRoom.name.label("room_name")
        )
        .join(PointRole, Alarm.role_id == PointRole.id)
        .join(FacilityUser, FacilityUser.facility_id == PointRole.facility_id)
        .join(Facility, Facility.id == FacilityUser.facility_id)
        .join(FacilityRoom, PointRole.room_id == FacilityRoom.id)
        .where(
            FacilityUser.user_id == user_id,
            PointRole.is_current == True
        )
        .order_by(Alarm.timestamp.desc())
    )

    result = await db.execute(stmt)
    rows = result.all()

    alarms_with_facility = []
    for alarm, facility_id, facility_name, room_id, room_name in rows:
        alarm_dict = {
            column.name: getattr(alarm, column.name)
            for column in alarm.__table__.columns
        }
        alarm_dict.update({
            'facility_id': facility_id,
            'facility_name': facility_name,
            'room_id': room_id,
            'room_name': room_name
        })
        alarms_with_facility.append(alarm_dict)

    return {'alarms': alarms_with_facility}

@router.get("/integrated")
async def get_integrated_posts(
    user_id: int, 
    page: int = Query(1, ge=1), 
    limit: int = Query(30, le=100),
    search_type: str = Query("all"), # all, title, creator, content
    keyword: Optional[str] = Query(None),
    db: AsyncSession = Depends(get_db)
):
    user_query = text("SELECT role FROM user_table WHERE id = :user_id")
    user_res = await db.execute(user_query, {"user_id": user_id})
    user = user_res.fetchone()

    if not user or user.role not in ['ADMIN', 'DEVELOPER']:
        raise HTTPException(status_code=403, detail="통합 조회 보드에 접근할 권한이 없습니다.")

    offset = (page - 1) * limit

    # 검색 쿼리 조건 생성
    where_clause = "WHERE 1=1"
    params = {"limit": limit, "offset": offset, "user_id": user_id}

    if keyword and keyword.strip():
        params["keyword"] = f"%{keyword.strip()}%"
        if search_type == "title":
            where_clause += " AND p.title ILIKE :keyword"
        elif search_type == "creator":
            where_clause += " AND u.name ILIKE :keyword"
        elif search_type == "content":
            where_clause += " AND p.content ILIKE :keyword"
        else: # "all"
            where_clause += " AND (p.title ILIKE :keyword OR u.name ILIKE :keyword OR p.content ILIKE :keyword)"

    # COUNT 쿼리
    count_query = text(f"""
        SELECT COUNT(*) 
        FROM posts p
        JOIN boards b ON p.board_id = b.board_id
        JOIN user_table u ON p.creator_id = u.id
        {where_clause}
    """)
    total_count = (await db.execute(count_query, params)).scalar()

    # Data 쿼리
    query = text(f"""
        SELECT p.post_id, p.title, p.content, p.creator_id, p.created_at,
               b.facility_id, b.facility_name, u.name as creator_name
        FROM posts p
        JOIN boards b ON p.board_id = b.board_id
        JOIN user_table u ON p.creator_id = u.id
        {where_clause}
        ORDER BY p.created_at DESC
        LIMIT :limit OFFSET :offset
    """)
    result = await db.execute(query, params)
    posts = result.fetchall()

    return {
        "status": "success",
        "data": [dict(row._mapping) for row in posts],
        "meta": {
            "page": page,
            "limit": limit,
            "total_count": total_count,
            "total_pages": (total_count + limit - 1) // limit
        }
    }

               b.facility_id, b.facility_name, u.name as creator_name
        FROM posts p
        JOIN boards b ON p.board_id = b.board_id
        JOIN user_table u ON p.creator_id = u.id
        WHERE p.post_id = :post_id
    """)
    post = (await db.execute(post_query, {"post_id": post_id})).fetchone()

    if not post:
        raise HTTPException(status_code=404, detail="포스트를 찾을 수 없습니다.")

    # 2. 권한 검증
    auth_query = text("""
        SELECT 1 FROM user_table u
        LEFT JOIN facility_users fu ON u.id = fu.user_id AND fu.facility_id = :facility_id
        WHERE u.id = :user_id AND (u.role IN ('ADMIN', 'DEVELOPER') OR fu.id IS NOT NULL)
    """)
    if not (await db.execute(auth_query, {"user_id": user_id, "facility_id": post.facility_id})).fetchone():
        raise HTTPException(status_code=403, detail="조회 권한이 없습니다.")

    comment_query = text("""
        SELECT c.comment_id, c.content, c.creator_id, c.created_at, u.name as creator_name
        FROM comments c
        JOIN user_table u ON c.creator_id = u.id
        WHERE c.post_id = :post_id
        ORDER BY c.created_at ASC
    """)
    comments = (await db.execute(comment_query, {"post_id": post_id})).fetchall()

    return {
        "status": "success",
        "data": dict(post._mapping),
        "comments": [dict(c._mapping) for c in comments]
    }

@router.get("/compare/{facility_id}/available-roles")

    """
    같은 시설 안에서 2개 이상의 호실에 공통으로 존재하는 role_name 목록을 반환한다.
    (예: 모든 냉동실에 있는 "고내온도" 센서를 한 번에 비교하기 위한 선택지)
    """
    # bool_ 타입은 꺾은선 그래프로 겹쳐 비교하기 적합하지 않아 목록에서 제외한다
    # (프론트의 MainChartViewer는 수치형 mean 값을 기준으로 그리기 때문).
    query = text("""
        SELECT pr.role_name, COUNT(DISTINCT pr.room_id) AS room_count,
               MAX(prm.display_name) AS display_name
        FROM point_role_table pr
        LEFT JOIN point_role_metadata prm ON pr.id = prm.role_id
        WHERE pr.facility_id = :facility_id AND pr.is_current = TRUE
          AND pr.role_name NOT ILIKE 'bool\\_%' ESCAPE '\\'

    """)
    rows = await db.execute(query, {"facility_id": facility_id})
    roles = [
        {
            "role_name": row.role_name,
            "display_name": row.display_name or row.role_name,
            "room_count": row.room_count,
        }
        for row in rows.fetchall()
    ]
    return {"status": "success", "data": {"roles": roles}}


@router.get("/compare/{facility_id}")
async def get_compare_data(
    facility_id: int,
    role_name: str = Query(..., description="호실 간 비교할 role_name"),
    scale: str = Query("1m", description="데이터 해상도"),
    time_range: str = Query("1d", description="조회 기간"),
    db: AsyncSession = Depends(get_db)
):
    """
    같은 시설의 여러 호실에서 동일한 role_name(예: 고내온도)을 가진 값들을
    호실별로 한 그래프에 겹쳐볼 수 있도록 반환한다.
    """
    if scale not in SCALE_MAPPING or time_range not in TIME_RANGE_MAPPING:
        raise HTTPException(status_code=400, detail="유효하지 않은 파라미터입니다.")

    bucket_interval = SCALE_MAPPING[scale]
    range_interval = TIME_RANGE_MAPPING[time_range]

    roles_query = text("""
        SELECT pr.id AS role_id, pr.room_id, fr.name AS room_name
        FROM point_role_table pr
        JOIN facilities_room_table fr ON fr.id = pr.room_id
        WHERE pr.facility_id = :facility_id AND pr.role_name = :role_name AND pr.is_current = TRUE
        ORDER BY fr.name
    """)
    role_rows = (await db.execute(roles_query, {"facility_id": facility_id, "role_name": role_name})).fetchall()

    if not role_rows:
        return {"status": "success", "data": {"datasets": []}, "message": "비교할 호실 데이터가 없습니다."}

    target_roles = [row.role_id for row in role_rows]
    room_by_role = {row.role_id: {"room_id": row.room_id, "room_name": row.room_name} for row in role_rows}

    r_name_lower = role_name.lower()
    if "temp_" in r_name_lower:
        table_name, agg_columns, data_key_mapping = "temperature_numerical_logs", "AVG(temp_mean) AS mean, MIN(temp_min) AS min, MAX(temp_max) AS max", {"mean": "mean", "min": "min", "max": "max"}
    elif "gear_" in r_name_lower:
        table_name, agg_columns, data_key_mapping = "gear_numerical_logs", "AVG(gear_mean) AS mean, MIN(gear_min) AS min, MAX(gear_max) AS max", {"mean": "mean", "min": "min", "max": "max"}
    elif "static_" in r_name_lower:
        table_name, agg_columns, data_key_mapping = "equipment_static_logs", "AVG(setnum) AS mean", {"mean": "mean"}
    elif "bool_" in r_name_lower:
        table_name, agg_columns, data_key_mapping = "bool_logs", "BOOL_OR(status_value) AS status", {"status": "status"}
    else:
        return {"status": "success", "data": {"datasets": []}, "message": "지원되지 않는 데이터 타입입니다."}

    max_time_query = text(f"SELECT MAX(timestamp) FROM {table_name} WHERE role_id = ANY(:role_ids)")
    latest_time = (await db.execute(max_time_query, {"role_ids": target_roles})).scalar()

    if not latest_time:
        return {"status": "success", "data": {"datasets": []}, "message": "기록된 로그가 없습니다."}

    datasets_map: Dict[int, Any] = {
        r_id: {
            "role_id": r_id,
            "room_id": room_by_role[r_id]["room_id"],
            "room_name": room_by_role[r_id]["room_name"],
            "role_name": role_name,
            "data": [],
        } for r_id in target_roles
    }

    query = text(f"""
        SELECT role_id, time_bucket('{bucket_interval}', timestamp) AS bucket, {agg_columns}
        FROM {table_name}
        WHERE role_id = ANY(:role_ids)
          AND timestamp >= (CAST(:latest_time AS timestamptz) - INTERVAL '{range_interval}')
          AND timestamp <= CAST(:latest_time AS timestamptz)
        GROUP BY role_id, bucket ORDER BY bucket ASC
    """)
    rows = await db.execute(query, {"role_ids": target_roles, "latest_time": latest_time})
    for r in rows.fetchall():
        r_dict = dict(r._mapping)
        point = {"timestamp": r_dict["bucket"].isoformat()}
        for db_key, out_key in data_key_mapping.items(): point[out_key] = r_dict[db_key]
        datasets_map[r_dict["role_id"]]["data"].append(point)

    return {
        "status": "success",
        "meta": {"facility_id": facility_id, "role_name": role_name, "scale": scale, "time_range": time_range, "reference_time": latest_time.isoformat()},
        "data": {"datasets": list(datasets_map.values())}
    }

    temp_roles, gear_roles, static_roles, bool_roles = [], [], [], []

    datasets_map: Dict[int, Any] = {

    async def fetch_and_map(table_name: str, agg_columns: str, type_roles: List[int], data_key_mapping: dict):

        query = text(f"""

class ConnectionManager:
    def __init__(self):
        self.active_connections: Dict[int, List[WebSocket]] = {}

    async def connect(self, websocket: WebSocket, room_id: int):
        await websocket.accept()
        if room_id not in self.active_connections:

        self.active_connections[room_id].append(websocket)

    def disconnect(self, websocket: WebSocket, room_id: int):
        if room_id in self.active_connections:
            self.active_connections[room_id].remove(websocket)
            if not self.active_connections[room_id]:
                del self.active_connections[room_id]

    async def broadcast_to_room(self, room_id: int, message: dict):
        if room_id in self.active_connections:
            for connection in self.active_connections[room_id]:
                await connection.send_text(json.dumps(message))

               (SELECT content FROM messages m WHERE m.room_id = cr.room_id ORDER BY created_at DESC LIMIT 1) as last_message,

                   JOIN user_table u ON cp2.user_id = u.id 
                   WHERE cp2.room_id = cr.room_id AND cp2.user_id != :user_id
                   LIMIT 1
               ) as other_user_role,
               (
                   SELECT COUNT(*)
                   FROM chatparticipants cp2
                   JOIN facility_users fu ON cp2.user_id = fu.user_id
                   WHERE cp2.room_id = cr.room_id AND cp2.user_id != :user_id AND fu.facility_id = :facility_id
               ) as other_user_in_facility
        FROM chatrooms cr
        JOIN chatparticipants cp ON cr.room_id = cp.room_id
        WHERE cp.user_id = :user_id
          AND (cr.type != 'FACILITY' OR cr.facility_id = :facility_id)
    """)

                if r.other_user_role in ['ADMIN', 'DEVELOPER']: priority = 2
                elif r.other_user_in_facility > 0: priority = 3
                else: priority = 6
            else:
                priority = 2 # 유저에게 관리자 1:1은 시설방 다음으로 중요
        elif r.type == 'GROUP':
            priority = 5 # (4는 즐겨찾기 용도로 비워둠)

        room_name_display = r.room_name
        if r.type == 'DIRECT' and not room_name_display:
            room_name_display = "1:1 채팅방"

        processed_rooms.append({
            "room_id": r.room_id,
            "room_name": room_name_display,
            "type": r.type,
            "last_message": r.last_message or "메시지가 없습니다.",
            "unread_count": r.unread_count,
            "sort_priority": priority,
            "last_message_time": str(r.last_message_time) if r.last_message_time else ""
        })

    processed_rooms.sort(key=lambda x: x["last_message_time"], reverse=True)
    processed_rooms.sort(key=lambda x: x["sort_priority"])

    return {"status": "success", "rooms": processed_rooms, "is_admin": is_admin}

    try:
        while True:
            data = await websocket.receive_text()
            payload = json.loads(data)
            
            msg_type = MessageType.IMAGE if payload.get("type") == "IMAGE" else MessageType.TEXT
            parent_id = payload.get("parent_message_id") # 답장 시 활용

            new_msg = Message(
                room_id=room_id, user_id=user_id, message_type=msg_type, 
                content=payload.get("content", ""), parent_message_id=parent_id
            )
            db.add(new_msg)
            await db.commit()
            await db.refresh(new_msg)
            
            update_unread = text("UPDATE chatparticipants SET unread_count = unread_count + 1 WHERE room_id = :room_id AND user_id != :user_id")
            await db.execute(update_unread, {"room_id": room_id, "user_id": user_id})
            await db.commit()

            parent_content = None
            if parent_id:
                parent_res = await db.execute(select(Message.content).where(Message.message_id == parent_id))
                parent_content = parent_res.scalar()

            broadcast_msg = {
                "type": "chat", "message_id": new_msg.message_id, "user_id": user_id,
                "message_type": msg_type.value, "content": new_msg.content,
                "created_at": new_msg.created_at.strftime("%Y-%m-%d %H:%M") if new_msg.created_at else "Just now",
                "parent_content": parent_content,
                "is_pinned": False
            }
            await manager.broadcast_to_room(room_id, broadcast_msg)
    except WebSocketDisconnect:
        manager.disconnect(websocket, room_id)

  <script>
    let cachedFacilities = [];

    async function loadFacilities() {
      try {
        const res = await fetch('/api/v1/serviceweb/admin/nodes/facilities');
        const json = await res.json();
        cachedFacilities = json.data || [];
        const select = document.getElementById('facSelect');
        select.innerHTML = '';

        if (cachedFacilities.length === 0) {
          select.innerHTML = '<option value="">등록된 시설이 없습니다</option>';
          return;
        }

        cachedFacilities.forEach(fac => {
          const opt = document.createElement('option');
          opt.value = fac.id;
          opt.textContent = fac.name;
          select.appendChild(opt);
        });
      } catch (err) {
        console.error("시설 목록 로드 실패:", err);
      }
    }

    async function loadNodes() {
      try {
        const res = await fetch('/api/v1/serviceweb/admin/nodes');
        const json = await res.json();
        const tbody = document.getElementById('nodeTableBody');
        tbody.innerHTML = '';

        if (!json.data || json.data.length === 0) {
          tbody.innerHTML = '<tr><td colspan="7" style="text-align: center; color: #64748b;">등록된 노드가 없습니다.</td></tr>';
          return;
        }

        json.data.forEach(node => {
          const tr = document.createElement('tr');
          const isOnline = node.status === 'ONLINE';
          tr.innerHTML = `
            <td>${node.id}</td>
            <td><strong>${node.facility_name || ('시설 #' + node.facility_id)}</strong></td>
            <td><strong>${node.node_code}</strong></td>
            <td>${node.node_name}</td>
            <td><span class="badge ${isOnline ? 'online' : 'offline'}">${node.status}</span></td>
            <td style="font-size: 12px; color: #94a3b8;">${node.last_heartbeat_at ? new Date(node.last_heartbeat_at).toLocaleTimeString() : '-'}</td>
            <td>
              <button class="btn-action btn-edit" onclick="editNode(${node.id}, '${node.node_name}')">수정</button>
              <button class="btn-action btn-rotate" onclick="rotateKey(${node.id})">재발급</button>
              <button class="btn-action btn-del" onclick="deleteNode(${node.id})">삭제</button>
            </td>
          `;
          tbody.appendChild(tr);
        });
      } catch (err) {
        console.error("노드 로드 실패:", err);
      }
    }

    async function editNode(nodeId, currentName) {
      const newName = prompt("변경할 노드 명칭을 입력하세요:", currentName);
      if (newName === null || newName.trim() === "" || newName === currentName) return;

      const res = await fetch(`/api/v1/serviceweb/admin/nodes/${nodeId}`, {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ node_name: newName.trim() })
      });
      if (res.ok) {
        loadNodes();
      } else {
        alert("수정 실패");
      }
    }

    async function createNode() {
      const facility_id = parseInt(document.getElementById('facSelect').value);
      const node_code = document.getElementById('nodeCode').value.trim();
      const node_name = document.getElementById('nodeName').value.trim();

      if (!facility_id || !node_code || !node_name) {
        alert("모든 필드를 입력하세요.");
        return;
      }

      const res = await fetch('/api/v1/serviceweb/admin/nodes', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ facility_id, node_code, node_name })
      });

      const json = await res.json();
      if (res.ok) {
        showKey(json.api_key);
        document.getElementById('nodeCode').value = '';
        document.getElementById('nodeName').value = '';
        loadNodes();
      } else {
        alert(json.detail || "생성 실패");
      }
    }

    async function rotateKey(nodeId) {
      if (!confirm("키를 재발급하면 기존 키로 연결된 통신은 즉시 차단됩니다. 진행하시겠습니까?")) return;
      const res = await fetch(`/api/v1/serviceweb/admin/nodes/${nodeId}/rotate-key`, { method: 'POST' });
      const json = await res.json();
      if (res.ok) {
        showKey(json.new_api_key);
      } else {
        alert(json.detail || "재발급 실패");
      }
    }

    async function deleteNode(nodeId) {
      if (!confirm("정말 이 노드를 삭제하시겠습니까? 관련 통신과 인증이 영구 차단됩니다.")) return;
      const res = await fetch(`/api/v1/serviceweb/admin/nodes/${nodeId}`, { method: 'DELETE' });
      if (res.ok) {
        loadNodes();
      } else {
        alert("삭제 실패");
      }
    }

    function showKey(key) {
      document.getElementById('modalApiKey').innerText = key;
      document.getElementById('keyModal').style.display = 'flex';
    }

    function closeModal() {
      navigator.clipboard.writeText(document.getElementById('modalApiKey').innerText);
      document.getElementById('keyModal').style.display = 'none';
      alert("키가 클립보드에 복사되었습니다.");
    }

    loadFacilities();
    loadNodes();
    setInterval(loadNodes, 3000);
