# React + TypeScript + Vite

This template provides a minimal setup to get React working in Vite with HMR and some Oxlint rules.

Currently, two official plugins are available:

- [@vitejs/plugin-react](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react) uses [Oxc](https://oxc.rs)
- [@vitejs/plugin-react-swc](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react-swc) uses [SWC](https://swc.rs/)

## React Compiler

The React Compiler is not enabled on this template because of its impact on dev & build performances. To add it, see [this documentation](https://react.dev/learn/react-compiler/installation).

## Expanding the Oxlint configuration

If you are developing a production application, we recommend enabling type-aware lint rules by installing `oxlint-tsgolint` and editing `.oxlintrc.json`:

```json
{

  },
  "rules": {
    "react/rules-of-hooks": "error",

```

import { useState } from 'react';
import { BrowserRouter as Router, Routes, Route, Navigate, NavLink } from 'react-router-dom';
import { ChartMappingPage } from './pages/ChartMappingPage';
import { ChartViewerPage }  from './pages/ChartViewerPage';
import { PRIndexingPage }    from './pages/PRIndexingPage';
import { FacilityManagerPage } from './pages/FacilityManagerPage';
import { RoomComparePage } from './pages/RoomComparePage';
import { LoginPage } from './pages/LoginPage';
import { MainPage } from './pages/MainPage';
import { Monogram, Wordmark } from './components/Brand';

function App() {
  const [user, setUser] = useState<any>(null); // 로그인한 유저 정보 상태

  const handleLogout = () => {
    setUser(null);
  };

  // 관리자 권한 체크용 상수
  const isAdmin = user && (user.role === 'ADMIN' || user.role === 'DEVELOPER');

  const navLinkStyle = ({ isActive }: { isActive: boolean }) =>
    `nav-link${isActive ? ' active' : ''}`;

  return (
    <Router>
      <div style={{ display: 'flex', flexDirection: 'column', height: '100vh', boxSizing: 'border-box' }}>

        {/* 헤더 */}
        <header
          style={{
            display: 'flex',
            justifyContent: 'space-between',
            alignItems: 'center',
            padding: '14px 24px',
            borderBottom: '1px solid var(--border)',
            background: 'var(--surface)',
          }}
        >
          <div style={{ display: 'flex', alignItems: 'center', gap: 10 }}>
            <Monogram size={32} />
            <Wordmark size={17} />
          </div>
          {user && (
            <div style={{ display: 'flex', alignItems: 'center', gap: 14 }}>
              <span style={{ fontSize: 13, color: 'var(--text-secondary)' }}>
                {user.name} <span className="chip" style={{ marginLeft: 6 }}>{user.role}</span>
              </span>
              <button onClick={handleLogout} className="btn btn-ghost btn-sm">로그아웃</button>
            </div>
          )}
        </header>

        {/* 글로벌 네비게이션 바 (로그인한 유저에게만 노출) */}
        {user && (
          <nav
            style={{
              display: 'flex', gap: 6, alignItems: 'center',
              padding: '10px 24px',
              borderBottom: '1px solid var(--border)',
              background: 'var(--surface)',
            }}
          >
            {isAdmin && (
              <>
                <NavLink to="/main" className={navLinkStyle}>메인 대시보드</NavLink>
                <NavLink to="/mapping" className={navLinkStyle}>맵핑 관리자</NavLink>
              </>
            )}

            {/* 유저와 관리자 공통 메뉴 */}
            <NavLink to="/viewer" className={navLinkStyle}>차트 뷰어</NavLink>

            {isAdmin && (
              <>
                <NavLink to="/indexing" className={navLinkStyle}>명칭 인덱싱</NavLink>
                <NavLink to="/facility-manager" className={navLinkStyle}>시설/호실 관리</NavLink>
                <NavLink to="/compare" className={navLinkStyle}>호실 비교</NavLink>
              </>
            )}
          </nav>
        )}

        {/* 본문 영역 */}
        <div style={{ flex: 1, display: 'flex', flexDirection: 'column', overflow: 'auto', background: 'var(--bg)' }}>
          <Routes>
            {/* 비로그인 상태일 때 접근 가능한 Route */}
            <Route path="/login" element={<LoginPage onLogin={setUser} />} />

            {/* 로그인 상태 체크 */}
            {!user ? (
              <Route path="*" element={<Navigate to="/login" />} />
            ) : (
              <>
                {/* Admin/Developer 전용 라우트 */}
                {isAdmin && (
                  <>
                    <Route path="/main" element={<MainPage />} />
                    <Route path="/mapping" element={<ChartMappingPage />} />
                    <Route path="/indexing" element={<PRIndexingPage />} />
                    <Route path="/facility-manager" element={<FacilityManagerPage />} />
                    <Route path="/compare" element={<RoomComparePage />} />
                  </>
                )}

                {/* User/Admin 공통 라우트 */}
                <Route path="/viewer" element={<ChartViewerPage />} />

                {/* 기본 리다이렉트 (관리자는 메인, 유저는 뷰어로) */}
                <Route path="*" element={
                  <Navigate to={isAdmin ? "/main" : "/viewer"} />
                } />
              </>
            )}
          </Routes>
        </div>
      </div>
    </Router>
  );
}

export default App;

const formatShortTime = (ts: number) => {
  const d = new Date(ts);
  return `${(d.getMonth() + 1).toString().padStart(2, '0')}/${d.getDate().toString().padStart(2, '0')} ${d.getHours().toString().padStart(2, '0')}:${d.getMinutes().toString().padStart(2, '0')}`;
};

const generatePostSegments = (validPosts: any[]) => {
  if (!validPosts || validPosts.length === 0) return [];
  const points = new Set<number>();
  validPosts.forEach(p => { points.add(p.st); points.add(p.et); });
  const sortedPoints = Array.from(points).sort((a, b) => a - b);
  const segments: any[] = [];
  
  for (let i = 0; i < sortedPoints.length - 1; i++) {
    const segSt = sortedPoints[i];
    const segEt = sortedPoints[i + 1];
    if (segSt === segEt) continue;
    const activePosts = validPosts.filter(p => p.st <= segSt && p.et >= segEt);
    if (activePosts.length > 0) segments.push({ st: segSt, et: segEt, activePosts });
  }
  return segments;
};

import React from 'react';

interface Props {
  datasets: any[];
  roleColors: Record<number, string>;
}

interface SeriesStat {
  roleId: number;
  name: string;
  color: string;
  min: number | null;
  avg: number | null;
  max: number | null;
}

const computeStats = (datasets: any[], roleColors: Record<number, string>): SeriesStat[] => {
  const stats: SeriesStat[] = [];

  for (const ds of datasets) {
    const points: any[] = ds.data || [];
    if (points.length === 0) continue;

    let sum = 0;
    let count = 0;
    let min: number | null = null;
    let max: number | null = null;

    for (const p of points) {
      const mean = typeof p.mean === 'number' ? p.mean : null;
      const pMin = typeof p.min === 'number' ? p.min : mean;
      const pMax = typeof p.max === 'number' ? p.max : mean;

      if (mean != null) { sum += mean; count++; }
      if (pMin != null) min = min == null ? pMin : Math.min(min, pMin);
      if (pMax != null) max = max == null ? pMax : Math.max(max, pMax);
    }

    if (count === 0 && min == null) continue;

    stats.push({
      roleId: ds.role_id,
      name: ds.display_name || ds.role_name || `시리즈 ${ds.role_id}`,
      color: roleColors[ds.role_id] || '#5b6b73',
      min,
      avg: count > 0 ? sum / count : null,
      max,
    });
  }

  return stats;
};

const fmt = (v: number | null) => (v == null ? '-' : v.toFixed(1));

/** 조회된 구간의 시리즈별 최소/평균/최대값 요약. 그래프를 눈으로 훑지 않아도 수치를 바로 확인할 수 있게 한다. */
export const ChartStatsPanel: React.FC<Props> = ({ datasets, roleColors }) => {
  const stats = computeStats(datasets, roleColors);
  if (stats.length === 0) return null;

  return (
    <div className="card" style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
      <h4 style={{ fontSize: 13, color: 'var(--text-secondary)', margin: 0 }}>구간 요약</h4>
      {stats.map(s => (
        <div key={s.roleId} style={{ display: 'flex', alignItems: 'center', gap: 8, fontSize: 12 }}>
          <span style={{ width: 8, height: 8, borderRadius: '50%', background: s.color, flexShrink: 0 }} />
          <span style={{ flex: 1, fontWeight: 600, overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{s.name}</span>
          <span style={{ color: 'var(--text-secondary)' }}>최소 {fmt(s.min)}</span>
          <span style={{ color: 'var(--text-secondary)' }}>평균 {fmt(s.avg)}</span>
          <span style={{ color: 'var(--text-secondary)' }}>최대 {fmt(s.max)}</span>
        </div>
      ))}
    </div>
  );
};

export const ChartMappingPage: React.FC = () => {
  const [facilities, setFacilities] = useState<any[]>([]);
  const [rooms, setRooms] = useState<any[]>([]);
  const [selectedFacility, setSelectedFacility] = useState<number | null>(null);
  const [selectedRoom, setSelectedRoom] = useState<number | null>(null);

  const [targetRoomRoles, setTargetRoomRoles] = useState<Role[]>([]);

  const [existingGroups, setExistingGroups] = useState<ChartGroup[]>([]);
  const [selectedGroupId, setSelectedGroupId] = useState<number | 'new'>('new');
  const [groupName, setGroupName] = useState('');
  const [droppedRoles, setDroppedRoles] = useState<Role[]>([]);

  useEffect(() => {
    apiService.getFacilities().then((data) => setFacilities(data.facilities || []));
  }, []);

  const handleFacilityChange = async (facId: number) => {
    setSelectedFacility(facId);
    setSelectedRoom(null);
    setTargetRoomRoles([]);
    setExistingGroups([]);
    resetGroupForm();

    const roomData = await apiService.getRooms(facId);
    setRooms(roomData.rooms || []);
    const roleData = await apiService.getPointRoles(facId, 0);
    setRoom0Roles(roleData.roles || []);
  };

  const handleRoomChange = async (roomId: number) => {
    setSelectedRoom(roomId);
    resetGroupForm();
    if (selectedFacility) {
      const roleData = await apiService.getPointRoles(selectedFacility, roomId);
      setTargetRoomRoles(roleData.roles || []);
      const groupData = await apiService.getChartGroups(selectedFacility, roomId);
      setExistingGroups(groupData.groups || []);
    }
  };

  const resetGroupForm = () => {
    setSelectedGroupId('new');
    setGroupName('');
    setDroppedRoles([]);
  };

  const handleGroupSelect = (val: string) => {
    if (val === 'new') {
      resetGroupForm();
      return;
    }
    const gId = Number(val);
    setSelectedGroupId(gId);
    const targetGroup = existingGroups.find((g) => g.group_id === gId);
    if (targetGroup) {
      setGroupName(targetGroup.group_name);
      const allAvailableRoles = [...room0Roles, ...targetRoomRoles];
      const validRoleIds = (targetGroup.role_ids || []).filter((id) => id !== null);
      const loadedRoles = validRoleIds
        .map((id) => allAvailableRoles.find((r) => r.role_id === id))
        .filter(Boolean) as Role[];
      setDroppedRoles(loadedRoles);
    }
  };

  const onDragStart = (e: React.DragEvent, role: Role) => {
    e.dataTransfer.setData('application/json', JSON.stringify(role));
  };

  const onDragOver = (e: React.DragEvent) => {
    e.preventDefault();
  };

  const onDrop = (e: React.DragEvent) => {
    e.preventDefault();
    const roleData = e.dataTransfer.getData('application/json');
    if (roleData) {
      const role: Role = JSON.parse(roleData);
      if (!droppedRoles.find((r) => r.role_id === role.role_id)) {
        setDroppedRoles([...droppedRoles, role]);
      }
    }
  };

  const handleReorderRoles = (sourceIndex: number, targetIndex: number) => {
    if (sourceIndex === targetIndex) return;
    const updated = [...droppedRoles];
    const [movedItem] = updated.splice(sourceIndex, 1);
    updated.splice(targetIndex, 0, movedItem);
    setDroppedRoles(updated);
  };

  const handleSaveGroup = async () => {
    if (!selectedFacility || !selectedRoom) return alert('시설과 호실을 선택하세요.');
    if (!groupName) return alert('그룹 명칭을 입력하세요.');
    if (droppedRoles.length === 0) return alert('포인트 롤을 추가하세요.');
    if (selectedGroupId !== 'new') return alert('수정 기능은 백엔드 구현 후 지원됩니다.');

    try {
      await apiService.createChartGroup({
        facility_id: selectedFacility,
        room_id: selectedRoom,
        group_name: groupName,
        role_ids: droppedRoles.map((r) => r.role_id),
      });
      alert('생성 완료!');
      handleRoomChange(selectedRoom);
    } catch (e) {
      alert('저장 실패');
    }
  };

  return (
    <div style={{ display: 'flex', flex: 1, minHeight: 0, height: '100%', gap: 12, padding: 24, boxSizing: 'border-box' }}>
      <LeftPanel
        facilities={facilities}
        rooms={rooms}
        selectedFacility={selectedFacility}
        selectedRoom={selectedRoom}
        room0Roles={room0Roles}
        targetRoomRoles={targetRoomRoles}
        droppedRoleIds={droppedRoles.map((r) => r.role_id)}
        onFacilityChange={handleFacilityChange}
        onRoomChange={handleRoomChange}
        onDragStart={onDragStart}
      />
      <RightPanel
        existingGroups={existingGroups}
        selectedGroupId={selectedGroupId}
        groupName={groupName}
        droppedRoles={droppedRoles}
        onGroupSelect={handleGroupSelect}
        onGroupNameChange={setGroupName}
        onDrop={onDrop}
        onDragOver={onDragOver}
        onRemoveRole={(id) => setDroppedRoles(droppedRoles.filter((r) => r.role_id !== id))}
        onReorderRoles={handleReorderRoles}
        onSaveGroup={handleSaveGroup}
      />
    </div>
  );
};

function useDebouncedCallback<T extends (...args: any[]) => void>(callback: T, delay: number) {
  const timeoutRef = useRef<number | null>(null);
  return useCallback((...args: Parameters<T>) => {
    if (timeoutRef.current) window.clearTimeout(timeoutRef.current);
    timeoutRef.current = window.setTimeout(() => callback(...args), delay);
  }, [callback, delay]);
}

    else if ((m = remain.match(dateRegex2))) { updatedDate.setMonth(parseInt(m[1]) - 1, parseInt(m[2])); remain = remain.replace(m[0], ' '); dateChanged = true; }

    const timeRegex1 = /(?:^|\s)([01]\d|2[0-3]):([0-5]\d)(?:\s|$)/; 
    const timeRegex2 = /(?:^|\s)([01]\d|2[0-3])([0-5]\d)(?:\s|$)/;  
    
    if ((m = remain.match(timeRegex1))) { updatedDate.setHours(parseInt(m[1]), parseInt(m[2]), 0, 0); remain = remain.replace(m[0], ' '); timeChanged = true; } 
    else if ((m = remain.match(timeRegex2))) { updatedDate.setHours(parseInt(m[1]), parseInt(m[2]), 0, 0); remain = remain.replace(m[0], ' '); timeChanged = true; }

    if (dateChanged && !timeChanged) updatedDate.setHours(0, 0, 0, 0);
    if (dateChanged || timeChanged) setTargetDate(getLocalDatetimeString(updatedDate));

    const q = remain.replace(/\s+/g, '');
    if (q && omniboxSuggestions.length > 0) handleOmniboxSelect(omniboxSuggestions[0]);
    else setOmniboxInput('');
  };

  const fetchAndMergeChunk = async (fetchStartMs: number, fetchEndMs: number, direction: 'replace' | 'append' | 'prepend') => {
    if (!selectedFacility || !selectedRoom || checkedRoles.length === 0) return;
    setIsChartLoading(true);
    try {
      const startIso = new Date(fetchStartMs).toISOString();
      const endIso = new Date(fetchEndMs).toISOString();
      const res = await apiService.getHistoricalData(Number(selectedFacility), Number(selectedRoom), selectedGroupId, scale, startIso, endIso, checkedRoles);
      
      const newDatasets = res.data?.datasets || [];
      const newPosts = res.data?.posts || [];
      const newAlarms = res.data?.alarms || [];
      const newHiddenPosts = res.data?.hidden_posts || [];

      if (direction === 'replace') {
        setChartDataSets(newDatasets); setPosts(newPosts); setAlarms(newAlarms); setHiddenPosts(newHiddenPosts);
        loadedRangeRef.current = { start: fetchStartMs, end: fetchEndMs };
        setCurrentViewStart(fetchStartMs);
      } else {
        setChartDataSets(prev => prev.map(prevDs => {
          const newDs = newDatasets.find((n: any) => n.role_id === prevDs.role_id);
          if (!newDs) return prevDs;
          let mergedData = direction === 'append' ? [...prevDs.data, ...newDs.data] : [...newDs.data, ...prevDs.data];
          mergedData.sort((a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime());
          if (mergedData.length > 5000) mergedData = direction === 'append' ? mergedData.slice(-5000) : mergedData.slice(0, 5000);
          return { ...prevDs, data: mergedData };
        }));

        setPosts(prev => { const combined = direction === 'append' ? [...prev, ...newPosts] : [...newPosts, ...prev]; return Array.from(new Map(combined.map(p => [p.id, p])).values()); });
        setAlarms(prev => { const combined = direction === 'append' ? [...prev, ...newAlarms] : [...newAlarms, ...prev]; return Array.from(new Map(combined.map(a => [a.alarm_id, a])).values()); });

        if (loadedRangeRef.current) {
          const config = CHART_SCOPING_CONFIG[mode][scale];
          const maxMsRange = config.chunkSize * config.maxChunks;
          let newStart = Math.min(loadedRangeRef.current.start, fetchStartMs);
          let newEnd = Math.max(loadedRangeRef.current.end, fetchEndMs);
          if (newEnd - newStart > maxMsRange) {
            if (direction === 'append') newStart = newEnd - maxMsRange;
            if (direction === 'prepend') newEnd = newStart + maxMsRange;
          }
          loadedRangeRef.current = { start: newStart, end: newEnd };
        }
      }
      setLoadedRangeState(loadedRangeRef.current);
    } catch (err) { console.error(err); } 
    finally { setIsChartLoading(false); }
  };

  useEffect(() => {
    if (selectedFacility && selectedRoom && checkedRoles.length > 0 && targetDate) {
      const endMs = new Date(targetDate).getTime();
      const chunkMs = CHART_SCOPING_CONFIG[mode][scale].chunkSize;
      fetchAndMergeChunk(endMs - chunkMs, endMs, 'replace');
    } else {
      setChartDataSets([]); setPosts([]); setAlarms([]); setHiddenPosts([]);
      loadedRangeRef.current = null; setLoadedRangeState(null); setCurrentViewStart(null);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [selectedFacility, selectedRoom, scale, targetDate, checkedRoles, refreshTrigger]); 

  const handleDataZoomFetch = useDebouncedCallback((viewStartMs: number, viewEndMs: number) => {
    if (isChartLoading || !loadedRangeRef.current || checkedRoles.length === 0) return;
    const config = CHART_SCOPING_CONFIG[mode][scale];
    const buffer = config.chunkSize * 0.2; 
    if (viewStartMs < loadedRangeRef.current.start + buffer) fetchAndMergeChunk(loadedRangeRef.current.start - config.chunkSize, loadedRangeRef.current.start, 'prepend');
    else if (viewEndMs > loadedRangeRef.current.end - buffer) fetchAndMergeChunk(loadedRangeRef.current.end, loadedRangeRef.current.end + config.chunkSize, 'append');
  }, 500);

  const handleZoomCapture = useCallback((start: number, end: number) => {
    setCurrentViewStart(start); 
    handleDataZoomFetch(start, end);
  }, [handleDataZoomFetch]);

  const isBoolRole = (roleId: number) => {
    const role = allRoles.find(r => String(r.role_id) === String(roleId));
    if (!role) return false;
    return role.data_type ? role.data_type === 'BOOL' : role.role_name.toLowerCase().includes('bool');
  }; 

  const handleModeToggle = () => {
    if (isChartLoading) return; 
    setMode(prev => {
      const nextMode = prev === 'DEFAULT' ? 'EXTENT' : 'DEFAULT';
      if (nextMode === 'EXTENT') {
        const boolRoles = checkedRoles.filter(id => isBoolRole(id));

        if (mainRoles.length > 1) setCheckedRoles([...boolRoles, mainRoles[0]]);
      }
      return nextMode;
    });
  }; 

  const handleRoleToggle = (roleId: number) => {
    if (isChartLoading) return;

    setCheckedRoles(prev => prev.includes(roleId) ? prev.filter(id => id !== roleId) : (mode === 'EXTENT' && !isBool ? [...prev.filter(id => isBoolRole(id)), roleId] : [...prev, roleId]));
  }; 

  const handleInvertToggle = (roleId: number) => setInvertedBools(prev => prev.includes(roleId) ? prev.filter(id => id !== roleId) : [...prev, roleId]);
  const handleColorChange = (roleId: number, color: string) => setRoleColors(prev => ({ ...prev, [roleId]: color }));
  const handleGuidelineToggle = (roleId: number) => setActiveGuidelines(prev => prev.includes(roleId) ? prev.filter(id => id !== roleId) : [...prev, roleId]);

  const syncChartDataZoom = (sourceRef: React.RefObject<ReactECharts | null>, targetRef: React.RefObject<ReactECharts | null>) => {
    const option = sourceRef.current?.getEchartsInstance().getOption() as any;

  };

    if (option?.dataZoom?.length > 0 && option.dataZoom[0].startValue) centerTime = (option.dataZoom[0].startValue + option.dataZoom[0].endValue) / 2;
    const zoomPayload = { type: 'dataZoom', startValue: centerTime - 1800000, endValue: centerTime + 1800000 };
    mainChartRef.current?.getEchartsInstance().dispatchAction(zoomPayload);
    subChartRef.current?.getEchartsInstance().dispatchAction(zoomPayload);
  };

    setTimeout(() => {
      const zoomPayload = { type: 'dataZoom', startValue: now.getTime() - (hours * 3600000), endValue: now.getTime() };
      mainChartRef.current?.getEchartsInstance().dispatchAction(zoomPayload);
      subChartRef.current?.getEchartsInstance().dispatchAction(zoomPayload);
    }, 300); 
  }; 

  const handleQuickDaySelect = (targetDayOfWeek: number) => {
    if (isChartLoading) return; 
    const now = new Date();
    let diff = now.getDay() - targetDayOfWeek;
    if (diff < 0) diff += 7;
    setTargetDate(getLocalDatetimeString(new Date(now.getTime() - (diff * 86400000))));

  };

  };

  };

      roles.forEach(r => {
        if (r.role_name.toLowerCase().includes(q) || r.display_name?.toLowerCase().includes(q) || r.short_name?.toLowerCase().includes(q)) {
          results.push({ type: 'role', id: r.role_id, label: `[롤] ${r.display_name || r.role_name}` });
        }
      });
    }
    setSuggestions(results.slice(0, 10));
  }, [searchInput, facilities, rooms, roles, selectedFacility, selectedRoom]);

  const handleApplySuggestion = (sug: any) => {
    if (sug.type === 'facility') handleFacilitySelect(sug.id);
    else if (sug.type === 'room') handleRoomSelect(sug.id);
    setSearchInput('');
    setSuggestions([]);
  };

  const handleSearchSubmit = () => {
    if (suggestions.length > 0) {
      handleApplySuggestion(suggestions[0]);
    }
  };

  const handleFieldChange = (roleId: number, field: 'display_name' | 'short_name', value: string) => {
    setEditMap(prev => ({
      ...prev,
      [roleId]: {
        ...prev[roleId],
        [field]: value
      }
    }));
  };

  // 전체 수정 저장
  const handleSaveAll = async () => {
    if (!selectedFacility || !selectedRoom) return alert('시설과 호실을 먼저 선택하세요.');

    const payload = roles.map(r => ({
      role_id: r.role_id,
      display_name: editMap[r.role_id]?.display_name || '',
      short_name: editMap[r.role_id]?.short_name || ''
    }));

    try {
      await apiService.updatePointRoleIndexing(payload);
      alert('인덱싱 명칭이 성공적으로 저장되었습니다.');
      handleRoomSelect(Number(selectedRoom)); // 새로고침
    } catch (e: any) {
      alert(`저장 실패: ${e.message}`);
    }
  };

  return (
    <div style={{ display: 'flex', flexDirection: 'column', flex: 1, minHeight: 0, gap: 12, padding: 24, boxSizing: 'border-box' }}>

      {/* 상단 컨트롤 바: 선택 + 검색 + 저장 */}
      <div className="card" style={{ display: 'flex', gap: 10, alignItems: 'center', padding: 14 }}>

        {/* 시설 / 호실 선택 드롭다운 */}
        <select
          value={selectedFacility}
          onChange={(e) => handleFacilitySelect(Number(e.target.value))}
          className="input"
          style={{ width: 180 }}
        >
          <option value="">-- 시설 선택 --</option>
          {facilities.map(f => <option key={f.id} value={f.id}>{f.name}</option>)}
        </select>

        <select
          value={selectedRoom}
          onChange={(e) => handleRoomSelect(Number(e.target.value))}
          disabled={!selectedFacility}
          className="input"
          style={{ width: 180 }}
        >
          <option value="">-- 호실 선택 --</option>
          {rooms.map(r => <option key={r.id} value={r.id}>{r.name}</option>)}
        </select>

        {/* 하향식 인덱싱 검색창 */}
        <div style={{ position: 'relative', flex: 1 }}>
          <input
            type="text"
            placeholder="시설 → 호실 → 포인트롤 검색 후 Enter"
            value={searchInput}
            onChange={(e) => setSearchInput(e.target.value)}
            onKeyDown={(e) => { if (e.key === 'Enter') handleSearchSubmit(); }}
            className="input"
          />
          {suggestions.length > 0 && (
            <div
              className="card"
              style={{ position: 'absolute', top: '110%', left: 0, right: 0, zIndex: 10, padding: 6 }}
            >
              {suggestions.map((sug, i) => (
                <div
                  key={i}
                  onClick={() => handleApplySuggestion(sug)}
                  style={{ padding: '8px 10px', cursor: 'pointer', borderRadius: 8, color: 'var(--primary)', fontSize: 13, fontWeight: 600 }}
                  onMouseEnter={(e) => (e.currentTarget.style.background = 'var(--primary-bg)')}
                  onMouseLeave={(e) => (e.currentTarget.style.background = 'transparent')}
                >
                  {sug.label}
                </div>
              ))}
            </div>
          )}
        </div>

        {/* 일괄 수정 확인 버튼 */}
        <button
          onClick={handleSaveAll}
          disabled={roles.length === 0 || isLoading}
          className="btn btn-primary"
        >
          수정 확인
        </button>
      </div>

      {/* 메인 인덱싱 테이블 */}
      <IndexingTable roles={roles} editMap={editMap} onFieldChange={handleFieldChange} />
    </div>
  );
};

      setRoleColors(colors);
      if (rawDatasets.length === 0) setErrorMsg(res.message || '조회된 데이터가 없습니다.');

        <div style={{ display: 'flex', flexDirection: 'column', gap: 4, minWidth: 180 }}>

        </div>
