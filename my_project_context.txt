@router.get("/my-context/{user_id}")
async def get_my_context(user_id: int, db: AsyncSession = Depends(get_db)):
    # 1. 유저 정보 조회
    user_query = text("SELECT id, email, name, role FROM user_table WHERE id = :user_id")
    user_result = await db.execute(user_query, {"user_id": user_id})
    user = user_result.fetchone()

    if not user:
        raise HTTPException(status_code=404, detail="User not found")

    # 2. 시설 정보 조회 로직
    # 관리자(ADMIN, DEVELOPER)는 모든 시설에 접근 가능
    if user.role in ['ADMIN', 'DEVELOPER']:
        facility_query = text("SELECT id, name FROM facilities_table WHERE is_active = TRUE")
        facility_result = await db.execute(facility_query)
    else:
        # 일반 유저(USER)는 facility_users 매핑 테이블에 존재하는 시설만 접근 가능
        facility_query = text("""
            SELECT f.id, f.name 
            FROM facilities_table f
            JOIN facility_users fu ON f.id = fu.facility_id
            WHERE fu.user_id = :user_id AND f.is_active = TRUE
        """)
        facility_result = await db.execute(facility_query, {"user_id": user_id})
        
    facilities = [{"facility_id": row.id, "facility_name": row.name} for row in facility_result.fetchall()]

    return {
        "status": "success",
        "user": {
            "id": user.id,
            "email": user.email,
            "name": user.name,
            "role": user.role
        },
        "accessible_facilities": facilities,
        # 임시로 클라이언트가 기본으로 선택할 첫 번째 시설 ID 반환 (없으면 None)
        "default_facility_id": facilities[0]["facility_id"] if facilities else None
    }

@router.get("/{user_id}", status_code=status.HTTP_200_OK)
async def get_user_facilities(user_id: int, db: AsyncSession = Depends(get_db)):
    """메인 색인 페이지 Box #2 및 시설 등록 페이지 Box #3용 시설 목록 조회"""
    # 1. 유저 권한 확인
    user_result = await db.execute(select(User).where(User.id == user_id))
    user = user_result.scalars().first()
    if not user:
        raise HTTPException(status_code=404, detail="사용자를 찾을 수 없습니다.")

    # 2. 권한별 시설 목록 반환
    if user.role in [UserRole.ADMIN, UserRole.DEVELOPER]:
        # 관리자/개발자: 모든 활성 시설 조회
        facilities_result = await db.execute(select(Facility).where(Facility.is_active == True))
        facilities = facilities_result.scalars().all()
    else:
        # 일반 유저: 매핑 테이블(facility_users)에 등록된 시설만 조회 (JOIN 대체 로직)
        mapping_result = await db.execute(select(FacilityUser.facility_id).where(FacilityUser.user_id == user_id))
        facility_ids = mapping_result.scalars().all()
        
        if not facility_ids:
            return {"facilities": []}
            
        facilities_result = await db.execute(select(Facility).where(Facility.id.in_(facility_ids), Facility.is_active == True))
        facilities = facilities_result.scalars().all()

    return {
        "facilities": [
            {"id": f.id, "name": f.name, "access_code": f.access_code} 
            for f in facilities
        ]
    }

@router.post("/{user_id}/register", status_code=status.HTTP_200_OK)
async def register_facility(user_id: int, req: RegisterFacilityRequest, db: AsyncSession = Depends(get_db)):
    """Box #2: 시설 접근 코드로 시설 등록"""
    # 1. 접근 코드로 시설 찾기
    fac_result = await db.execute(select(Facility).where(Facility.access_code == req.access_code, Facility.is_active == True))
    facility = fac_result.scalars().first()
    
    if not facility:
        raise HTTPException(status_code=404, detail="유효하지 않거나 비활성화된 접근 코드입니다.")

    # 2. 중복 등록 확인
    mapping_result = await db.execute(
        select(FacilityUser).where(FacilityUser.facility_id == facility.id, FacilityUser.user_id == user_id)
    )
    if mapping_result.scalars().first():
        raise HTTPException(status_code=400, detail="이미 등록된 시설입니다.")

    # 3. 매핑 추가
    new_mapping = FacilityUser(facility_id=facility.id, user_id=user_id)
    db.add(new_mapping)
    await db.commit()

    return {"message": "시설이 성공적으로 등록되었습니다.", "facility_name": facility.name}

@router.delete("/{user_id}/unlink/{facility_id}", status_code=status.HTTP_200_OK)
async def unlink_facility(user_id: int, facility_id: int, db: AsyncSession = Depends(get_db)):
    """Box #3: 시설 연동 해제"""
    mapping_result = await db.execute(
        select(FacilityUser).where(FacilityUser.facility_id == facility_id, FacilityUser.user_id == user_id)
    )
    mapping = mapping_result.scalars().first()
    
    if not mapping:
        raise HTTPException(status_code=404, detail="해당 시설에 등록되어 있지 않습니다.")

    await db.delete(mapping)
    await db.commit()
    return {"message": "시설 연동이 해제되었습니다."}

@router.post("/signup/general", status_code=status.HTTP_201_CREATED)

        email=user_in.email,
        password=hashed_password,
        name=user_in.name, # 프런트의 '아이디(필수)'에 해당
        phone_number=user_in.phone_number,
        role=UserRole.USER 
    )
    db.add(new_user)
    await db.commit()
    await db.refresh(new_user)
    
    return {"message": "일반 계정이 성공적으로 생성되었습니다.", "user_id": new_user.id}

@router.post("/find-id", status_code=status.HTTP_200_OK)
async def find_id(req: FindIdRequest, db: AsyncSession = Depends(get_db)):
    """Box #5: 아이디(이메일) 찾기"""
    
    result = await db.execute(
        select(User).where(
            User.name == req.name,
            User.phone_number == req.phone_number
        )
    )
    user = result.scalars().first()

    if not user:
        raise HTTPException(status_code=404, detail="일치하는 계정 정보가 없습니다.")

    # 보안을 위해 이메일 일부 마스킹 처리 (예: test***@gmail.com)
    email_parts = user.email.split("@")
    masked_id = email_parts[0][:3] + "***" if len(email_parts[0]) > 3 else email_parts[0][:1] + "***"
    masked_email = f"{masked_id}@{email_parts[1]}"

    return {"message": "아이디 찾기 성공", "email": masked_email}

def hash_code(code: str) -> str:
    """인증 코드를 해싱합니다."""
    return hashlib.sha256(code.encode()).hexdigest()

async def request_password_code(req: PasswordRequestCode, db: AsyncSession = Depends(get_db)):

@router.post("/password/verify-code", status_code=status.HTTP_200_OK)
async def verify_password_code(req: PasswordVerifyCode):
    """Box #2: 입력 코드 확인 및 재설정 토큰 발급"""
    try:
        payload = jwt.decode(req.token, SECRET_KEY, algorithms=[ALGORITHM])
        email = payload.get("sub")
        code_hash = payload.get("code_hash")

        # 클라이언트가 보낸 코드를 해싱하여 토큰의 해시값과 비교
        if hash_code(req.code) != code_hash:
            raise HTTPException(status_code=400, detail="코드가 일치하지 않습니다.")

        # 인증 성공 시, 비밀번호 변경 전용 새 JWT 발급 (만료시간 10분)
        expire = datetime.now(timezone.utc) + timedelta(minutes=10)
        reset_token = jwt.encode(
            {"sub": email, "action": "reset_pw", "exp": expire}, 
            SECRET_KEY, 
            algorithm=ALGORITHM
        )

        return {"message": "코드 인증이 완료되었습니다.", "reset_token": reset_token}
        
    except jwt.ExpiredSignatureError:
        raise HTTPException(status_code=400, detail="인증 시간이 초과되었습니다. 코드를 다시 요청해주세요.")
    except jwt.InvalidTokenError:
        raise HTTPException(status_code=400, detail="유효하지 않은 토큰입니다.")

@router.get("/integrated")
async def get_integrated_posts(
    user_id: int, 
    page: int = Query(1, ge=1), 
    limit: int = Query(30, le=100), 
    db: AsyncSession = Depends(get_db)
):
    user_query = text("SELECT role FROM user_table WHERE id = :user_id")
    user_res = await db.execute(user_query, {"user_id": user_id})
    user = user_res.fetchone()

    if not user or user.role not in ['ADMIN', 'DEVELOPER']:
        raise HTTPException(status_code=403, detail="통합 조회 보드에 접근할 권한이 없습니다.")

    offset = (page - 1) * limit
    
    # COUNT 쿼리 (클라이언트에서 전체 페이지 계산을 위함)
    count_query = text("SELECT COUNT(*) FROM posts")
    total_count = (await db.execute(count_query)).scalar()

    query = text("""
        SELECT p.post_id, p.title, p.content, p.creator_id, p.created_at,
               b.facility_id, b.facility_name, u.name as creator_name
        FROM posts p
        JOIN boards b ON p.board_id = b.board_id
        JOIN user_table u ON p.creator_id = u.id
        ORDER BY p.created_at DESC
        LIMIT :limit OFFSET :offset
    """)
    result = await db.execute(query, {"limit": limit, "offset": offset})
    posts = result.fetchall()

    return {
        "status": "success",
        "data": [dict(row._mapping) for row in posts],
        "meta": {
            "page": page,
            "limit": limit,
            "total_count": total_count,
            "total_pages": (total_count + limit - 1) // limit
        }
    }

@router.post("/post")
async def create_post(user_id: int, req: PostCreateReq, db: AsyncSession = Depends(get_db)):
    # 1. 포스트 삽입
    query = text("""
        INSERT INTO posts (board_id, creator_id, updater_id, title, content)
        VALUES (:board_id, :user_id, :user_id, :title, :content)
        RETURNING post_id
    """)
    result = await db.execute(query, {
        "board_id": req.board_id, 
        "user_id": user_id, 
        "title": req.title, 
        "content": req.content
    })
    new_post_id = result.scalar()

    # 2. 선(Pre) 업로드된 이미지 ID 맵핑 (임시로 post_id=NULL 이었던 것을 업데이트)
    if req.image_ids:
        mapping_query = text("""
            UPDATE post_images 
            SET post_id = :post_id 
            WHERE image_id = ANY(:image_ids)
        """)
        await db.execute(mapping_query, {"post_id": new_post_id, "image_ids": req.image_ids})
        
    await db.commit()
    return {"status": "success", "post_id": new_post_id}

        raise HTTPException(status_code=403, detail="수정 권한이 없습니다.")

    # 2. 포스트 업데이트
    update_query = text("""
        UPDATE posts 
        SET title = COALESCE(:title, title),
            content = COALESCE(:content, content),
            updater_id = :user_id
        WHERE post_id = :post_id
    """)
    await db.execute(update_query, {
        "title": req.title, 
        "content": req.content, 
        "user_id": user_id, 
        "post_id": post_id
    })

    # 3. 추가된 이미지가 있다면 맵핑
    if req.image_ids:
        mapping_query = text("""
            UPDATE post_images 
            SET post_id = :post_id 
            WHERE image_id = ANY(:image_ids)
        """)
        await db.execute(mapping_query, {"post_id": post_id, "image_ids": req.image_ids})

    await db.commit()
    return {"status": "success", "message": "포스트 수정 완료"}

@router.get("/{user_id}", status_code=status.HTTP_200_OK)

    """메인 색인 페이지 Box #2 및 시설 등록 페이지 Box #3용 시설 목록 조회"""
    # 1. 유저 권한 확인
    user_result = await db.execute(select(User).where(User.id == user_id))
    user = user_result.scalars().first()
    if not user:
        raise HTTPException(status_code=404, detail="사용자를 찾을 수 없습니다.")

    # 2. 권한별 시설 목록 반환
    if user.role in [UserRole.ADMIN, UserRole.DEVELOPER]:
        # 관리자/개발자: 모든 활성 시설 조회
        facilities_result = await db.execute(select(Facility).where(Facility.is_active == True))
        facilities = facilities_result.scalars().all()
    else:
        # 일반 유저: 매핑 테이블(facility_users)에 등록된 시설만 조회 (JOIN 대체 로직)
        mapping_result = await db.execute(select(FacilityUser.facility_id).where(FacilityUser.user_id == user_id))
        facility_ids = mapping_result.scalars().all()
        
        if not facility_ids:
            return {"facilities": []}
            
        facilities_result = await db.execute(select(Facility).where(Facility.id.in_(facility_ids), Facility.is_active == True))
        facilities = facilities_result.scalars().all()

    return {
        "facilities": [
            {"id": f.id, "name": f.name, "access_code": f.access_code} 
            for f in facilities
        ]
    }

async def register_facility(user_id: int, req: RegisterFacilityRequest, db: AsyncSession = Depends(get_db)):
    """Box #2: 시설 접근 코드로 시설 등록"""
    # 1. 접근 코드로 시설 찾기
    fac_result = await db.execute(select(Facility).where(Facility.access_code == req.access_code, Facility.is_active == True))
    facility = fac_result.scalars().first()
    
    if not facility:
        raise HTTPException(status_code=404, detail="유효하지 않거나 비활성화된 접근 코드입니다.")

    # 2. 중복 등록 확인
    mapping_result = await db.execute(
        select(FacilityUser).where(FacilityUser.facility_id == facility.id, FacilityUser.user_id == user_id)
    )
    if mapping_result.scalars().first():
        raise HTTPException(status_code=400, detail="이미 등록된 시설입니다.")

    # 3. 매핑 추가
    new_mapping = FacilityUser(facility_id=facility.id, user_id=user_id)
    db.add(new_mapping)
    await db.commit()

    return {"message": "시설이 성공적으로 등록되었습니다.", "facility_name": facility.name}

from sqlalchemy import Column, Integer, String, Enum, DateTime, func, ForeignKey
from sqlalchemy.orm import relationship
from config.db import Base
# ===============================================================

class UserRole(str, enum.Enum):
    USER = "USER"
    ADMIN = "ADMIN"
    DEVELOPER = "DEVELOPER"


class User(Base):
    __tablename__ = "user_table"

    id = Column(Integer, primary_key=True, autoincrement=True)
    email = Column(String(255), unique=True, nullable=False, index=True)
    password = Column(String(255), nullable=True)
    name = Column(String(100), nullable=False)
    role = Column(Enum(UserRole, name="user_role"), nullable=False, default=UserRole.USER)
    phone_number = Column(String(20), nullable=True)
    
    # 타임존을 포함하는 생성일, 수정일 (자동 갱신)
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    updated_at = Column(DateTime(timezone=True), server_default=func.now(), onupdate=func.now())

    external_providers = relationship("ExternalProvider", back_populates="user", cascade="all, delete-orphan")


    # 역참조 맵핑 
    facilities = relationship(
        "Facility",
        secondary="facility_users",
        primaryjoin="User.id == facility_users.c.user_id",
        secondaryjoin="Facility.id == facility_users.c.facility_id",
        viewonly=True # 다대다 조회 목적
    )


class ExternalProvider(Base):
    __tablename__ = "external_providers"

    external_provider_id = Column(Integer, primary_key=True, autoincrement=True)
    account_id = Column(Integer, ForeignKey("user_table.id", ondelete="CASCADE"), nullable=False)
    provider = Column(String(50), nullable=False) # 'google' or 'apple' 제한
    provider_id = Column(String(255), nullable=False)
    created_at = Column(DateTime(timezone=True), server_default=func.now())

    user = relationship("User", back_populates="external_providers")

final GoRouter appRouter = GoRouter(
  initialLocation: '/',
  errorBuilder: (context, state) => const NotFoundScreen(),

  // refreshListenable: authService, // 인증 상태 변화 감지 (AuthService 연결 필요)
  redirect: (context, state) {
    final bool loggedIn = AuthService.isLoggedIn;
    final String path = state.matchedLocation;

    // 비로그인 상태에서도 허용할 경로 리스트 (오프닝, 로그인, 가입, 아이디/비번 찾기)
    final allowedPaths = ['/', '/login', '/signup', '/find_id', '/find_pw'];

    // 로그인하지 않은 사용자가 허용되지 않은 경로로 접근하려고 할 때만 로그인 페이지로 리다이렉트
    if (!loggedIn && !allowedPaths.contains(path)) {
      return '/login';
    }
    return null;
  },

  routes: [
    // [common]
    GoRoute(path: '/', builder: (context, state) => const OpeningPage()),
    GoRoute(path: '/login', builder: (context, state) => const LoginPage()),
    GoRoute(path: '/signup', builder: (context, state) => const SignupPage()),
    GoRoute(path: '/find_id', builder: (context, state) => const FindIdPage()),
    GoRoute(path: '/find_pw', builder: (context, state) => const FindPwPage()),
    // GoRoute(path: '/reset_pw', builder: (context, state) => const ResetPwPage()),

    // [main_indices] - 유저 ID 기반
    GoRoute(path: '/main_index/:userId', builder: (context, state) => MainIndexPage(userId: state.pathParameters['userId']!)),
    GoRoute(path: '/facility_register/:userId', builder: (context, state) => FacilityRegisterPage(userId: state.pathParameters['userId']!)),
    GoRoute(path: '/main_alarm/:userId', builder: (context, state) => MainAlarmPage(userId: state.pathParameters['userId']!)),
    // GoRoute(path: '/total_board/:userId', builder: (context, state) => TotalBoardPage(userId: state.pathParameters['userId']!)),
    // main_indices/[settings]
    GoRoute(path: '/settings/:userId', builder: (context, state) => SettingsPage(userId: state.pathParameters['userId']!)),
    GoRoute(path: '/personal/:userId', builder: (context, state) => PersonalPage(userId: state.pathParameters['userId']!)),

    // [facility] - 유저 ID & 시설 ID 기반
    GoRoute(
      path: '/facility_nav/:userId/:facilityId',
      builder: (context, state) => FacilityTempNavPage(
        userId: state.pathParameters['userId']!,
        facilityId: state.pathParameters['facilityId']!,

import 'dart:async';
import 'package:flutter/material.dart';
import 'package:go_router/go_router.dart';

import 'package:idcs/services/common/find_pw/auth_api_service.dart';

class FindPwPage extends StatefulWidget {
  const FindPwPage({super.key});

  @override
  State<FindPwPage> createState() => _FindPwPageState();
}

class _FindPwPageState extends State<FindPwPage> {
  // 단계 제어 (1: 이메일, 2: 코드 입력, 3: 새 비밀번호 설정)
  int _currentStep = 1;

  // 상태 변수
  String _jwtToken = '';       // Box 1에서 발급받은 코드 확인용 JWT
  String _resetToken = '';     // Box 2에서 발급받은 비밀번호 변경용 JWT
  String _box1Message = '';
  String _box2Message = '';
  String _box3Message = '';

  // 컨트롤러
  final _emailController = TextEditingController();
  final _codeController = TextEditingController();
  final _pwController = TextEditingController();
  final _pwConfirmController = TextEditingController();

  // 타이머 관련
  Timer? _timer;
  int _remainingSeconds = 300; // 5분

  @override
  void dispose() {
    _timer?.cancel();
    _emailController.dispose();
    _codeController.dispose();
    _pwController.dispose();
    _pwConfirmController.dispose();
    super.dispose();
  }

  void _startTimer() {
    _timer?.cancel();
    _remainingSeconds = 300;
    _timer = Timer.periodic(const Duration(seconds: 1), (timer) {
      setState(() {
        if (_remainingSeconds > 0) {
          _remainingSeconds--;
        } else {
          _timer?.cancel();
          _box2Message = '입력 시간이 초과되었습니다.';
        }
      });
    });
  }

  String _formatTime(int seconds) {
    final m = (seconds / 60).floor();
    final s = seconds % 60;
    return '${m.toString().padLeft(2, '0')}:${s.toString().padLeft(2, '0')}';
  }

  // Box #1 제출 로직
  Future<void> _handleEmailSubmit() async {
    final email = _emailController.text.trim();
    if (email.isEmpty) return;

    try {
      final res = await AuthApiService.requestPasswordCode(email);
      setState(() {
        _jwtToken = res['token'];
        // 개발용 콘솔에만 찍히도록 백엔드에서 준 dev_code 표시 가능 (테스트용)
        _box1Message = res['message'] + " (Test Code: ${res['dev_code']})";
        _currentStep = 2; // Box 2 열기
      });
      _startTimer();
    } catch (e) {
      setState(() => _box1Message = e.toString());
    }
  }

  // Box #2 제출 로직
  Future<void> _handleCodeSubmit() async {
    final code = _codeController.text.trim();
    if (code.isEmpty || _remainingSeconds <= 0) return;

    try {
      final res = await AuthApiService.verifyPasswordCode(_jwtToken, code);
      setState(() {
        _resetToken = res['reset_token'];
        _box2Message = '코드 인증 성공!';
        _timer?.cancel();
        _currentStep = 3; // Box 3 열기
      });
    } catch (e) {
      setState(() => _box2Message = e.toString());
    }
  }

  // Box #3 제출 로직
  Future<void> _handlePasswordReset() async {
    final pw1 = _pwController.text;
    final pw2 = _pwConfirmController.text;

    if (pw1.isEmpty || pw2.isEmpty) return;
    if (pw1 != pw2) {
      setState(() => _box3Message = '비밀번호가 일치하지 않습니다.');
      return;
    }

    try {
      final res = await AuthApiService.resetPassword(_resetToken, pw1);
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(res['message'])));
      context.go('/login'); // 완료 시 로그인 페이지 이동
    } catch (e) {
      setState(() => _box3Message = e.toString());
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('비밀번호 찾기 페이지')),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            // Box #1: 이메일 입력
            _buildBoxContainer(
              isActive: true,
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.stretch,
                children: [
                  _buildInputRow('이메일 입력', _emailController, isEnabled: _currentStep == 1),
                  const SizedBox(height: 12),
                  ElevatedButton(
                    onPressed: _currentStep == 1 ? _handleEmailSubmit : null,
                    child: const Text('버튼(submit) 연동 계정 확인하기'),
                  ),
                  if (_box1Message.isNotEmpty)
                    Padding(
                      padding: const EdgeInsets.only(top: 8.0),
                      child: Text(_box1Message, style: const TextStyle(color: Colors.purple)),
                    ),
                ],
              ),
            ),
            const SizedBox(height: 16),

            // Box #2: 코드 입력 (AnimatedSize로 Collapse 효과 부여)
            AnimatedSize(
              duration: const Duration(milliseconds: 300),
              child: _currentStep >= 2
                  ? _buildBoxContainer(
                isActive: _currentStep == 2,
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.stretch,
                  children: [
                    _buildInputRow('코드 입력', _codeController, isEnabled: _currentStep == 2),
                    const SizedBox(height: 12),
                    ElevatedButton(
                      onPressed: _currentStep == 2 ? _handleCodeSubmit : null,
                      child: const Text('버튼(submit) 입력 코드 확인'),
                    ),
                    const SizedBox(height: 8),
                    Text('남은 시간: ${_formatTime(_remainingSeconds)}',
                        style: const TextStyle()),
                    if (_box2Message.isNotEmpty)
                      Padding(
                        padding: const EdgeInsets.only(top: 8.0),
                        child: Text(_box2Message, style: const TextStyle()),
                      ),
                  ],
                ),
              )
                  : const SizedBox.shrink(), // 단계가 아니면 보이지 않음
            ),
            SizedBox(height: _currentStep >= 2 ? 16 : 0),

            // Box #3: 새 비밀번호 입력
            AnimatedSize(
              duration: const Duration(milliseconds: 300),
              child: _currentStep >= 3
                  ? _buildBoxContainer(
                isActive: _currentStep == 3,
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.stretch,
                  children: [
                    _buildInputRow('새 비밀번호', _pwController, isObscure: true),
                    const SizedBox(height: 8),
                    _buildInputRow('재입력', _pwConfirmController, isObscure: true),
                    const SizedBox(height: 12),
                    if (_box3Message.isNotEmpty)
                      Padding(
                        padding: const EdgeInsets.only(bottom: 8.0),
                        child: Text(_box3Message, style: const TextStyle()),
                      ),
                    ElevatedButton(
                      onPressed: _handlePasswordReset,
                      child: const Text('버튼(submit) 비밀번호 다시 설정하기'),
                    ),
                  ],
                ),
              )
                  : const SizedBox.shrink(),
            ),
          ],
        ),
      ),
    );
  }

  // 공통 UI 빌더 헬퍼
  Widget _buildBoxContainer({required Widget child, required bool isActive}) {
    return Container(
      decoration: BoxDecoration(
        border: Border.all(),
      ),
      padding: const EdgeInsets.all(12.0),
      // 비활성 박스는 반투명 처리
      child: Opacity(opacity: isActive ? 1.0 : 0.5, child: child),
    );
  }

  Widget _buildInputRow(String label, TextEditingController controller, {bool isObscure = false, bool isEnabled = true}) {
    return Row(
      children: [
        SizedBox(width: 80, child: Text(label, style: const TextStyle(fontSize: 13))),
        Expanded(
          child: TextField(
            controller: controller,
            obscureText: isObscure,
            enabled: isEnabled,
            decoration: const InputDecoration(
              border: OutlineInputBorder(),
              isDense: true,
            ),
          ),
        ),
      ],
    );
  }

  Future<void> _handleGeneralLogin() async {

    if (email.isEmpty || password.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('이메일과 비밀번호를 모두 입력해주세요.')),
      );
      return;

        email: email,
        password: password,
        keepLogin: _keepLogin,
      );

      // 1. AuthService에 발급받은 user_id 저장 (이 시점에 isLoggedIn이 true가 됨)

      AuthService.setUserId(userId);

      // [비동기 컨텍스트 방어 로직] 화면이 파괴되지 않았는지 확인
      if (!mounted) return;

      );

      AuthService.setUserId(userId);

      if (!mounted) return;

        SnackBar(content: Text(result['message'])),
      );

      context.go('/main_index/$userId');

    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(e.toString())),
      );
    }
  }

class _SignupPageState extends State<SignupPage> {
  final TextEditingController _emailController = TextEditingController();
  final TextEditingController _passwordController = TextEditingController();
  final TextEditingController _nameController = TextEditingController();
  final TextEditingController _phoneController = TextEditingController();

  // --- Box 3 (애플 폴백) 관련 상태 변수 ---
  bool _showAppleFallback = false;
  String _pendingAppleProviderId = '';
  String _pendingAppleName = '';
  final TextEditingController _fallbackEmailController = TextEditingController();

  @override
  void dispose() {
    _emailController.dispose();
    _passwordController.dispose();
    _nameController.dispose();
    _phoneController.dispose();
    super.dispose();
  }

  // API 연동: 일반 계정 생성 처리
  Future<void> _handleGeneralSignup() async {
    final email = _emailController.text.trim();
    final password = _passwordController.text.trim();
    final name = _nameController.text.trim();
    final phone = _phoneController.text.trim();

    if (email.isEmpty || password.isEmpty || name.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('필수 항목을 모두 입력해주세요.')),
      );
      return;
    }

    try {
      final result = await AuthApiService.signupGeneral(
        email: email,
        password: password,
        name: name,
        phoneNumber: phone,
      );

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(result['message'])),
      );
      // 가입 성공 시 로그인 페이지로 이동
      context.go('/login');
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(e.toString())),
      );
    }
  }

  // API 연동: 소셜 계정 생성 처리
  Future<void> _handleSocialSignup(String provider) async {
    try {
      Map<String, String>? socialData;

      if (provider == 'google') {
        socialData = await SocialAuthService.signInWithGoogle();
      } else if (provider == 'apple') {
        socialData = await SocialAuthService.signInWithApple();
      }

      if (socialData == null) return;

      // 이메일 누락 방어 로직 (Box 3 트리거)
      if (socialData['email'] == null || socialData['email']!.isEmpty) {
        if (provider == 'apple') {
          setState(() {
            _showAppleFallback = true; // Box 3 열기
            _pendingAppleProviderId = socialData!['provider_id']!;
            _pendingAppleName = socialData['name']!;

        } else {

        }

      await _submitSocialSignup(
        provider: socialData['provider']!,
        providerId: socialData['provider_id']!,
        email: socialData['email']!,
        name: socialData['name']!,
      );

      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(e.toString())));

  Future<void> _handleAppleFallbackSubmit() async {

import 'dart:convert';
import 'package:http/http.dart' as http;
import '../../main.dart';

class FacilityApiService {
  static String get _baseUrl => '${AppConfig.baseUrl}/facilities';

  // 시설 목록 조회 (권한에 따라 백엔드에서 알아서 필터링됨)
  static Future<List<dynamic>> getFacilities(String userId) async {
    final url = Uri.parse('$_baseUrl/$userId');
    final response = await http.get(url);
    if (response.statusCode == 200) {
      final data = jsonDecode(utf8.decode(response.bodyBytes));
      return data['facilities'] ?? [];
    } else {
      throw Exception('시설 목록을 불러오는데 실패했습니다.');
    }
  }

  // 시설 등록
  static Future<String> registerFacility(String userId, String accessCode) async {
    final url = Uri.parse('$_baseUrl/$userId/register');
    final response = await http.post(
      url,
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({'access_code': accessCode}),
    );
    final data = jsonDecode(utf8.decode(response.bodyBytes));
    if (response.statusCode == 200) return data['message'];
    throw Exception(data['detail'] ?? '시설 등록에 실패했습니다.');
  }

  // 시설 연동 해제
  static Future<String> unlinkFacility(String userId, int facilityId) async {
    final url = Uri.parse('$_baseUrl/$userId/unlink/$facilityId');
    final response = await http.delete(url);
    final data = jsonDecode(utf8.decode(response.bodyBytes));
    if (response.statusCode == 200) return data['message'];
    throw Exception(data['detail'] ?? '시설 해제에 실패했습니다.');
  }
}

          rethrow; // 다른 종류의 에러는 그대로 던집니다.
        }
      }
      final googleUser = await googleSignIn.authenticate();
      // 사용자가 로그인 팝업을 그냥 닫은 경우 null 반환
      if (googleUser == null) {
        return null;
      }

      return {
        'provider': 'google',
        'provider_id': googleUser.id,
        'email': googleUser.email,
        'name': googleUser.displayName ?? '구글유저',
      };
    } catch (error) {
      throw Exception('구글 로그인 진행 중 오류가 발생/취소되었습니다: $error');
    }
  }

  /// 애플 SDK를 통해 프로필 정보를 가져옵니다.
  static Future<Map<String, String>?> signInWithApple() async {
    try {
      final AuthorizationCredentialAppleID credential =
      await SignInWithApple.getAppleIDCredential(
        scopes: [
          AppleIDAuthorizationScopes.email,
          AppleIDAuthorizationScopes.fullName,
        ],
      );

      String name = '애플유저';
      if (credential.givenName != null || credential.familyName != null) {
        name = '${credential.familyName ?? ''}${credential.givenName ?? ''}';
      }

      return {
        'provider': 'apple',
        'provider_id': credential.userIdentifier ?? '',
        'email': credential.email ?? '',
        'name': name,
      };
    } catch (error) {
      throw Exception('애플 로그인 진행 중 오류가 발생/취소되었습니다: $error');
    }
  }
